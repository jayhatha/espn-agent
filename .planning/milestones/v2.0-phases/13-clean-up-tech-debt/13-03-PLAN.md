  <action>
    Add dead letter queue to capture permanently failed jobs after max retries.

    Current behavior (lines 38-67): Failed event listener logs error but doesn't persist failure data.

    1. Create DLQ queue at module level:
    ```typescript
    import { Queue, Worker, type Job } from "bullmq";
    import { connection } from "../db/connection.js";

    export const snapshotQueue = new Queue("snapshots", { connection });
    export const snapshotDLQ = new Queue("snapshots-dlq", { connection });
    ```

    2. Update failed event handler (line 38):
    ```typescript
    worker.on("failed", async (job, err) => {
      if (job && job.attemptsMade >= (job.opts.attempts || 3)) {
        // Max retries exceeded - move to DLQ
        logger.warn({ jobId: job.id }, "Max retries exceeded - moving to DLQ");

        await snapshotDLQ.add("failed-snapshot", {
          originalJobId: job.id,
          originalJobName: job.name,
          originalData: job.data,
          errorMessage: err.message,
          errorStack: err.stack,
          attemptsMade: job.attemptsMade,
          failedAt: new Date().toISOString()
        }, {
          removeOnComplete: { age: 86400 * 30 } // Keep 30 days for analysis
        });

        logger.error({ jobId: job.id, error: err }, "Job moved to DLQ");
      }
    });
    ```

    3. Add DLQ inspection helper (optional, for admin debugging):
    ```typescript
    export async function inspectDLQ(limit = 10) {
      const failed = await snapshotDLQ.getJobs(["completed"], 0, limit);
      return failed.map(job => ({
        jobId: job.data.originalJobId,
        failedAt: job.data.failedAt,
        error: job.data.errorMessage,
        attempts: job.data.attemptsMade
      }));
    }
    ```

    Fix two critical worker stability issues.

    **Issue 1: morning-report-worker calls process.exit(0) (lines 12-18)**

    Problem: When Slack is disabled, worker calls process.exit(0) which kills entire process including co-located workers.

    Fix:
    ```typescript
    // BEFORE
    if (!isSlackEnabled()) {
      logger.info("Slack not configured - exiting morning report worker");
      process.exit(0);
    }

    // AFTER
    if (!isSlackEnabled()) {
      logger.warn("Slack disabled - morning report worker inactive");
      // Don't start this worker, but let process continue
      return;
    }
    ```

    **Issue 2: player-pool-worker uses node-cron (lines 15-48)**

    Problem: node-cron has no retry, no persistence, no deduplication across instances. Should use BullMQ repeatable jobs instead.

    Fix (migrate to BullMQ):
    ```typescript
    import { Queue, Worker } from "bullmq";
    import { connection } from "../db/connection.js";

    // Create queue with repeatable job pattern
    export const playerPoolQueue = new Queue("player-pool-sync", { connection });

    // Schedule daily job at 4 AM ET
    await playerPoolQueue.add(
      "daily-sync",
      {},
      {
        repeat: {
          pattern: "0 4 * * *", // 4 AM daily
          tz: "America/New_York"
        },
        removeOnComplete: { age: 86400 * 7 }, // Keep 7 days
        removeOnFail: { age: 86400 * 30 }
      }
    );

    // Worker to process jobs
    const worker = new Worker(
      "player-pool-sync",
      async (job) => {
        // Existing sync logic from node-cron callback
        const agent = new PlayerPoolSyncAgent(db, logger);
        await agent.syncPlayerPool();
      },
      {
        connection,
        attempts: 3,
        backoff: { type: "exponential", delay: 5000 }
      }
    );
    ```

    Remove node-cron dependency after migration.

    Add missing UNIQUE constraints to prevent duplicate data.

    **Issue 1: refresh_tokens.token_hash has no UNIQUE constraint (migration 008)**
    Permits duplicate token hashes - security issue.

    **Issue 2: transaction_impact constraint too narrow (migration 009:48)**
    Only includes player_gained_id, not player_lost_id - drop-only transactions can duplicate.

    Create migration 014:
    ```sql
    -- Migration 014: Add missing unique constraints
    -- Created: 2026-02-26

    -- Add UNIQUE constraint on refresh_tokens.token_hash
    ALTER TABLE refresh_tokens
    ADD CONSTRAINT refresh_tokens_token_hash_unique UNIQUE (token_hash);

    -- Drop old transaction_impact constraint and replace with composite
    ALTER TABLE transaction_impact
    DROP CONSTRAINT IF EXISTS transaction_impact_unique_per_transaction_player;

    ALTER TABLE transaction_impact
    ADD CONSTRAINT transaction_impact_unique_per_transaction
    UNIQUE (league_id, transaction_date, transaction_type, player_gained_id, player_lost_id);

    -- Note: This composite constraint handles all transaction types:
    -- - Adds: player_gained_id set, player_lost_id NULL
    -- - Drops: player_gained_id NULL, player_lost_id set
    -- - Trades: both set
    ```

    No schema.ts or types.ts changes needed - constraints are database-level only.
