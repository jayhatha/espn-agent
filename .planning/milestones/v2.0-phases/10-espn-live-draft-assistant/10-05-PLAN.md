**1. Create AuctionBudgetTracker component (components/AuctionBudgetTracker.tsx):**

```typescript
import React, { useMemo } from 'react';
import { useDraftStore } from '../lib/draft-state';
import { calculateBudgetPacing } from '../lib/value-calculations';

export default function AuctionBudgetTracker() {
  const { myRoster, remainingBudget, rosterSize, draftFormat } = useDraftStore();

  const pacing = useMemo(() => {
    // Assume standard auction budget is 260
    const totalBudget = 260;
    return calculateBudgetPacing(totalBudget, myRoster, rosterSize);
  }, [myRoster, rosterSize]);

  if (draftFormat !== 'auction') {
    return null; // Only show for auction drafts
  }

  const pacingColors = {
    ahead: 'bg-green-50 border-green-200 text-green-900',
    'on-track': 'bg-blue-50 border-blue-200 text-blue-900',
    behind: 'bg-red-50 border-red-200 text-red-900',
  };

  const pacingLabels = {
    ahead: '✓ Ahead of pace (can be aggressive)',
    'on-track': '→ On track',
    behind: '⚠ Behind pace (conserve budget)',
  };

  return (
    <div className="p-3 border-b bg-gray-50">
      <h3 className="font-semibold text-sm mb-2">Budget Pacing</h3>

      {/* Budget overview */}
      <div className="bg-white p-3 rounded border mb-2">
        <div className="flex justify-between mb-2">
          <span className="text-sm text-gray-600">Remaining Budget</span>
          <span className="text-lg font-bold text-gray-900">
            ${pacing.remainingBudget}
          </span>
        </div>

        <div className="flex justify-between mb-2">
          <span className="text-sm text-gray-600">Roster Spots Left</span>
          <span className="font-semibold text-gray-900">
            {pacing.remainingRosterSpots}
          </span>
        </div>

        <div className="flex justify-between">
          <span className="text-sm text-gray-600">Avg per Player</span>
          <span className="font-semibold text-gray-900">
            ${Math.round(pacing.averageRemainingPerPlayer)}
          </span>
        </div>
      </div>

      {/* Pacing status */}
      <div className={`p-2 rounded border text-sm ${pacingColors[pacing.pacingStatus]}`}>
        {pacingLabels[pacing.pacingStatus]}
      </div>

      {/* Recommended max bid */}
      <div className="mt-2 bg-blue-50 p-2 rounded border border-blue-200">
        <div className="text-xs text-blue-700 mb-1">Recommended Max Bid</div>
        <div className="text-xl font-bold text-blue-900">
          ${pacing.recommendedMaxBid}
        </div>
        <div className="text-xs text-blue-600 mt-1">
          (40% of remaining budget)
        </div>
      </div>
    </div>
  );
}
```

**1. Create AuctionPlayerCard component (components/AuctionPlayerCard.tsx):**

```typescript
import React from 'react';
import type { AuctionValue } from '@espn-fantasy/shared';
import { useDraftStore } from '../lib/draft-state';
import { calculateBudgetPacing } from '../lib/value-calculations';

interface AuctionPlayerCardProps {
  player: AuctionValue;
  rank: number;
}

export default function AuctionPlayerCard({ player, rank }: AuctionPlayerCardProps) {
  const { myRoster, rosterSize } = useDraftStore();

  // Calculate if player is affordable
  const pacing = calculateBudgetPacing(260, myRoster, rosterSize);
  const isAffordable = player.dollarValue <= pacing.recommendedMaxBid;
  const isValuePick = player.dollarValue <= pacing.averageRemainingPerPlayer;

  return (
    <div className="p-3 border-b hover:bg-blue-50 cursor-pointer transition-colors border-gray-200">
      {/* Player name and positions */}
      <div className="flex justify-between items-start mb-1">
        <div>
          <div className="font-semibold text-gray-900">{player.playerName}</div>
          <div className="text-xs text-gray-500">
            {player.positions.join(', ')}
          </div>
        </div>
        <div className="text-right">
          <div className="text-lg font-bold text-green-600">
            ${player.dollarValue}
          </div>
          <div className="text-xs text-gray-500">
            VBD: {player.vbdScore.toFixed(1)}
          </div>
        </div>
      </div>

      {/* Value indicators */}
      <div className="flex gap-2 mt-2">
        {isValuePick && (
          <span className="text-xs bg-green-100 text-green-700 px-2 py-1 rounded">
            ⭐ Value
          </span>
        )}
        {!isAffordable && (
          <span className="text-xs bg-red-100 text-red-700 px-2 py-1 rounded">
            Over budget
          </span>
        )}
      </div>

      {/* Projected stats (abbreviated) */}
      <div className="text-xs text-gray-600 mt-2 flex gap-3">
        {player.projectedStats.HR && <span>HR: {player.projectedStats.HR}</span>}
        {player.projectedStats.RBI && <span>RBI: {player.projectedStats.RBI}</span>}
        {player.projectedStats.SB && <span>SB: {player.projectedStats.SB}</span>}
        {player.projectedStats.W && <span>W: {player.projectedStats.W}</span>}
        {player.projectedStats.SV && <span>SV: {player.projectedStats.SV}</span>}
      </div>
    </div>
  );
}
```

**2. Create AuctionRecommendations component (components/AuctionRecommendations.tsx):**

```typescript
import React, { useMemo } from 'react';
import { useDraftStore } from '../lib/draft-state';
import AuctionPlayerCard from './AuctionPlayerCard';
import { calculateBudgetPacing } from '../lib/value-calculations';

export default function AuctionRecommendations() {
  const {
    auctionValues,
    selectedPosition,
    setPositionFilter,
    myRoster,
    rosterSize,
    allPicks,
  } = useDraftStore();

  // Calculate budget pacing for filtering
  const pacing = useMemo(() => {
    return calculateBudgetPacing(260, myRoster, rosterSize);
  }, [myRoster, rosterSize]);

  // Filter out already-drafted players
  const draftedPlayerIds = new Set(allPicks.map(p => p.playerId));
  const availablePlayers = useMemo(() => {
    return auctionValues.filter(p => !draftedPlayerIds.has(p.playerId));
  }, [auctionValues, allPicks]);

  // Apply filters
  const filteredPlayers = useMemo(() => {
    let players = availablePlayers;

    if (selectedPosition) {
      players = players.filter(p => p.positions.includes(selectedPosition));
    }

    // Sort by dollar value descending
    return players.sort((a, b) => b.dollarValue - a.dollarValue);
  }, [availablePlayers, selectedPosition]);

  // Categorize players by affordability
  const affordablePlayers = filteredPlayers.filter(
    p => p.dollarValue <= pacing.recommendedMaxBid
  );
  const allPlayers = filteredPlayers;

  const positions = ['C', '1B', '2B', '3B', 'SS', 'OF', 'SP', 'RP'];

  return (
    <div className="flex flex-col h-full">
      {/* Filter controls */}
      <div className="p-3 bg-gray-50 border-b space-y-2">
        <div className="flex gap-1 flex-wrap">
          <button
            onClick={() => setPositionFilter(null)}
            className={`
              px-2 py-1 text-xs rounded
              ${!selectedPosition ? 'bg-blue-600 text-white' : 'bg-white text-gray-700 border'}
            `}
          >
            All
          </button>
          {positions.map(pos => (
            <button
              key={pos}
              onClick={() => setPositionFilter(pos)}
              className={`
                px-2 py-1 text-xs rounded
                ${selectedPosition === pos ? 'bg-blue-600 text-white' : 'bg-white text-gray-700 border'}
              `}
            >
              {pos}
            </button>
          ))}
        </div>

        <div className="text-xs text-gray-600">
          {affordablePlayers.length} affordable (≤${pacing.recommendedMaxBid})
        </div>
      </div>

      {/* Player list */}
      <div className="flex-1 overflow-y-auto">
        {allPlayers.length === 0 ? (
          <div className="p-4 text-center text-gray-500">
            {auctionValues.length === 0 ? 'Loading values...' : 'No players match filters'}
          </div>
        ) : (
          allPlayers.slice(0, 50).map((player, index) => (
            <AuctionPlayerCard
              key={player.playerId}
              player={player}
              rank={index + 1}
            />
          ))
        )}
      </div>
    </div>
  );
}
```
