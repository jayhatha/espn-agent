// Add to packages/extension/lib/value-calculations.ts

export interface PickValue {
  pickNumber: number;
  playerId: number;
  playerName: string;
  teamName: string;
  expectedRank: number;
  actualPick: number;
  value: number; // Positive = value (picked later than expected), negative = reach (picked earlier)
  valueLabel: string; // e.g., "5-pick value" or "8-pick reach"
}

export function calculatePickValue(
  pick: DraftPick,
  rankings: DraftRanking[],
): PickValue {
  // Find player in rankings to get expected rank
  const playerRanking = rankings.find(r => r.playerId === pick.playerId);

  if (!playerRanking) {
    return {
      pickNumber: pick.pickNumber,
      playerId: pick.playerId,
      playerName: pick.playerName,
      teamName: pick.teamName,
      expectedRank: 0,
      actualPick: pick.pickNumber,
      value: 0,
      valueLabel: 'Unranked',
    };
  }

  // Find expected rank based on VBD score
  const sortedRankings = [...rankings].sort((a, b) => b.vbdScore - a.vbdScore);
  const expectedRank = sortedRankings.findIndex(r => r.playerId === pick.playerId) + 1;

  // Value = expected rank - actual pick
  // Positive = picked later than expected (good value)
  // Negative = picked earlier than expected (reach/overpay)
  const value = expectedRank - pick.pickNumber;

  let valueLabel: string;
  if (value > 0) {
    valueLabel = `${value}-pick value`;
  } else if (value < 0) {
    valueLabel = `${Math.abs(value)}-pick reach`;
  } else {
    valueLabel = 'On target';
  }

  return {
    pickNumber: pick.pickNumber,
    playerId: pick.playerId,
    playerName: pick.playerName,
    teamName: pick.teamName,
    expectedRank,
    actualPick: pick.pickNumber,
    value,
    valueLabel,
  };
}

// Calculate value for all picks
export function calculateAllPickValues(
  picks: DraftPick[],
  rankings: DraftRanking[],
): PickValue[] {
  return picks.map(pick => calculatePickValue(pick, rankings));
}

// Create packages/extension/components/PickValueTracker.tsx

import React, { useMemo } from 'react';
import { useDraftStore } from '../lib/draft-state';
import { calculateAllPickValues } from '../lib/value-calculations';

export default function PickValueTracker() {
  const { allPicks, rankings, myTeamId } = useDraftStore();

  const pickValues = useMemo(() => {
    return calculateAllPickValues(allPicks, rankings);
  }, [allPicks, rankings]);

  // Group by team
  const picksByTeam = useMemo(() => {
    const grouped: Record<string, typeof pickValues> = {};
    pickValues.forEach(pv => {
      if (!grouped[pv.teamName]) {
        grouped[pv.teamName] = [];
      }
      grouped[pv.teamName].push(pv);
    });
    return grouped;
  }, [pickValues]);

  // Calculate team efficiency (average value per pick)
  const teamEfficiency = useMemo(() => {
    return Object.entries(picksByTeam).map(([teamName, picks]) => {
      const avgValue = picks.reduce((sum, p) => sum + p.value, 0) / picks.length;
      return { teamName, avgValue, pickCount: picks.length };
    }).sort((a, b) => b.avgValue - a.avgValue);
  }, [picksByTeam]);

  if (allPicks.length === 0) {
    return (
      <div className="p-3 text-sm text-gray-500 text-center">
        No picks yet
      </div>
    );
  }

  return (
    <div className="flex flex-col h-full">
      {/* Team efficiency leaderboard */}
      <div className="p-3 bg-gray-50 border-b">
        <h3 className="font-semibold text-sm mb-2">Draft Efficiency</h3>
        <div className="space-y-1">
          {teamEfficiency.slice(0, 5).map((team, index) => (
            <div
              key={team.teamName}
              className={`flex justify-between text-xs p-2 rounded ${
                team.teamName === myTeamId ? 'bg-blue-100 border border-blue-300' : 'bg-white border'
              }`}
            >
              <span className="font-medium">
                {index + 1}. {team.teamName}
              </span>
              <span className={team.avgValue > 0 ? 'text-green-600' : team.avgValue < 0 ? 'text-red-600' : 'text-gray-600'}>
                {team.avgValue > 0 ? '+' : ''}{team.avgValue.toFixed(1)} avg
              </span>
            </div>
          ))}
        </div>
      </div>

      {/* Recent picks with value indicators */}
      <div className="flex-1 overflow-y-auto">
        <div className="p-3">
          <h3 className="font-semibold text-sm mb-2">Recent Picks</h3>
          <div className="space-y-2">
            {pickValues.slice(-10).reverse().map(pv => (
              <div
                key={pv.pickNumber}
                className={`p-2 rounded border ${
                  pv.teamName === myTeamId ? 'bg-blue-50 border-blue-200' : 'bg-white'
                }`}
              >
                <div className="flex justify-between items-start mb-1">
                  <div className="text-sm font-medium">
                    Pick {pv.pickNumber}: {pv.playerName}
                  </div>
                  <div
                    className={`text-xs px-2 py-1 rounded ${
                      pv.value > 3
                        ? 'bg-green-100 text-green-700'
                        : pv.value < -3
                        ? 'bg-red-100 text-red-700'
                        : 'bg-gray-100 text-gray-600'
                    }`}
                  >
                    {pv.valueLabel}
                  </div>
                </div>
                <div className="text-xs text-gray-500">
                  {pv.teamName} • Expected rank: #{pv.expectedRank}
                </div>
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}

// Create packages/extension/components/PreDraftRankingEditor.tsx

import React, { useState, useMemo } from 'react';
import { useDraftStore } from '../lib/draft-state';

export default function PreDraftRankingEditor() {
  const { rankings, userRankingOverrides, setPlayerOverrideRank, clearPlayerOverride, clearAllOverrides } = useDraftStore();
  const [searchQuery, setSearchQuery] = useState('');

  // Apply user overrides to rankings
  const adjustedRankings = useMemo(() => {
    const withOverrides = rankings.map(player => ({
      ...player,
      userRank: userRankingOverrides[player.playerId] || null,
      isOverridden: player.playerId in userRankingOverrides,
    }));

    // Sort by user rank if exists, otherwise VBD score
    return withOverrides.sort((a, b) => {
      const rankA = a.userRank || withOverrides.findIndex(p => p.playerId === a.playerId) + 1;
      const rankB = b.userRank || withOverrides.findIndex(p => p.playerId === b.playerId) + 1;
      return rankA - rankB;
    });
  }, [rankings, userRankingOverrides]);

  // Filter by search
  const filteredRankings = useMemo(() => {
    if (!searchQuery) return adjustedRankings;
    const query = searchQuery.toLowerCase();
    return adjustedRankings.filter(p =>
      p.playerName.toLowerCase().includes(query) ||
      p.positions.some(pos => pos.toLowerCase().includes(query))
    );
  }, [adjustedRankings, searchQuery]);

  const handleRankChange = (playerId: number, newRank: string) => {
    const rank = parseInt(newRank, 10);
    if (isNaN(rank) || rank < 1) {
      clearPlayerOverride(playerId);
    } else {
      setPlayerOverrideRank(playerId, rank);
    }
  };

  const overrideCount = Object.keys(userRankingOverrides).length;

  return (
    <div className="flex flex-col h-full">
      {/* Header */}
      <div className="p-3 bg-gray-50 border-b">
        <h3 className="font-semibold text-sm mb-2">Customize Rankings</h3>
        <p className="text-xs text-gray-600 mb-3">
          Adjust player ranks before draft starts. Your changes will affect recommendations.
        </p>

        {/* Search */}
        <input
          type="text"
          placeholder="Search players..."
          value={searchQuery}
          onChange={(e) => setSearchQuery(e.target.value)}
          className="w-full px-3 py-2 text-sm border rounded mb-2"
        />

        {/* Override stats */}
        <div className="flex justify-between items-center">
          <span className="text-xs text-gray-600">
            {overrideCount} player{overrideCount !== 1 ? 's' : ''} customized
          </span>
          {overrideCount > 0 && (
            <button
              onClick={clearAllOverrides}
              className="text-xs text-red-600 hover:text-red-700"
            >
              Clear all
            </button>
          )}
        </div>
      </div>

      {/* Player list */}
      <div className="flex-1 overflow-y-auto">
        {filteredRankings.slice(0, 100).map((player, index) => (
          <div
            key={player.playerId}
            className={`p-3 border-b ${
              player.isOverridden ? 'bg-yellow-50 border-yellow-200' : 'hover:bg-gray-50'
            }`}
          >
            <div className="flex justify-between items-start gap-3">
              <div className="flex-1">
                <div className="font-semibold text-sm">{player.playerName}</div>
                <div className="text-xs text-gray-500">
                  {player.positions.join(', ')} • VBD: {player.vbdScore.toFixed(1)}
                </div>
              </div>

              <div className="flex items-center gap-2">
                <div className="text-xs text-gray-500">Rank:</div>
                <input
                  type="number"
                  min="1"
                  placeholder={`${index + 1}`}
                  value={player.userRank || ''}
                  onChange={(e) => handleRankChange(player.playerId, e.target.value)}
                  className="w-16 px-2 py-1 text-sm border rounded text-center"
                />
                {player.isOverridden && (
                  <button
                    onClick={() => clearPlayerOverride(player.playerId)}
                    className="text-xs text-red-600 hover:text-red-700"
                  >
                    Reset
                  </button>
                )}
              </div>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}

interface PlayerCardProps {
  player: DraftRanking;
  rank: number;
  isAtTurn?: boolean;
}

export default function PlayerCard({ player, rank, isAtTurn }: PlayerCardProps) {
  const { currentPick, myTeamId, totalTeams, allPicks, draftFormat } = useDraftStore();

  // Calculate picks until user's next turn
  const picksUntilMyTurn = useMemo(() => {
    if (!myTeamId || draftFormat !== 'snake') return null;

    // Find user's last pick
    const myPicks = allPicks.filter(p => p.teamId === myTeamId);
    const lastPickNumber = myPicks.length > 0 ? Math.max(...myPicks.map(p => p.pickNumber)) : 0;

    // In snake draft, calculate next pick based on round
    const currentRound = Math.floor((currentPick - 1) / totalTeams) + 1;
    const positionInRound = ((currentPick - 1) % totalTeams) + 1;

    // Simplified: assume user's pick position is constant (not accounting for snake reversal)
    // Full implementation would track actual pick order
    const picksUntil = totalTeams - positionInRound;

    return picksUntil > 0 ? picksUntil : null;
  }, [currentPick, myTeamId, totalTeams, allPicks, draftFormat]);

  return (
    <div
      className={`
        p-3 border-b hover:bg-blue-50 cursor-pointer transition-colors
        ${isAtTurn ? 'bg-yellow-50 border-yellow-200' : 'border-gray-200'}
      `}
    >
      {/* Player name and positions */}
      <div className="flex justify-between items-start mb-1">
        <div>
          <div className="font-semibold text-gray-900">{player.playerName}</div>
          <div className="text-xs text-gray-500 space-x-2">
            <span>{player.positions.join(', ')} • Tier {player.tier}</span>
            {player.adp && (
              <span className="text-blue-600">ADP: {player.adp.toFixed(1)}</span>
            )}
          </div>
        </div>
        <div className="text-right">
          <div className="text-sm font-bold text-blue-600">#{rank}</div>
          <div className="text-xs text-gray-500">VBD: {player.vbdScore.toFixed(1)}</div>
        </div>
      </div>

      {/* Picks until turn indicator */}
      {picksUntilMyTurn !== null && picksUntilMyTurn <= 5 && (
        <div className="mt-1 text-xs bg-orange-100 text-orange-700 px-2 py-1 rounded inline-block">
          {picksUntilMyTurn} pick{picksUntilMyTurn !== 1 ? 's' : ''} until your turn
        </div>
      )}

      {/* Projected stats (abbreviated) */}
      <div className="text-xs text-gray-600 mt-2 flex gap-3">
        {player.projectedStats.HR && <span>HR: {player.projectedStats.HR}</span>}
        {player.projectedStats.RBI && <span>RBI: {player.projectedStats.RBI}</span>}
        {player.projectedStats.SB && <span>SB: {player.projectedStats.SB}</span>}
        {player.projectedStats.AVG && <span>AVG: {player.projectedStats.AVG.toFixed(3)}</span>}
        {player.projectedStats.W && <span>W: {player.projectedStats.W}</span>}
        {player.projectedStats.SV && <span>SV: {player.projectedStats.SV}</span>}
        {player.projectedStats.ERA && <span>ERA: {player.projectedStats.ERA.toFixed(2)}</span>}
      </div>
    </div>
  );
}
