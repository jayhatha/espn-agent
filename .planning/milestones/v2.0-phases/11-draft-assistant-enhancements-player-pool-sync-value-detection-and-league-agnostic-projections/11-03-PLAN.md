    ```typescript
    import type { Category } from "@espn-fantasy/shared";

    export interface DraftConfiguration {
      categories: Category[];
      rosterSlots: Record<string, number>;
      totalTeams: number;
    }

    export interface DetectionResult {
      config: DraftConfiguration | null;
      confidence: "high" | "low" | "failed";
      warnings: string[];
    }

    /**
     * Detect league configuration from ESPN draft page DOM
     * User constraint: Auto-detect from DOM, fallback to 5x5 defaults on failure
     */
    export function detectLeagueConfig(): DetectionResult {
      const warnings: string[] = [];

      // 1. Detect total teams
      const totalTeams = detectTeamCount();
      if (!totalTeams) {
        warnings.push("Could not detect team count from draft board");
      }

      // 2. Detect scoring categories
      const categories = detectCategories();
      if (!categories || categories.length === 0) {
        warnings.push("Could not detect scoring categories from league settings");
      }

      // 3. Detect roster slots
      const rosterSlots = detectRosterSlots();
      if (!rosterSlots || Object.keys(rosterSlots).length === 0) {
        warnings.push("Could not detect roster slots from league settings");
      }

      // If any detection failed, return low confidence or failed
      if (!totalTeams || !categories || !rosterSlots) {
        return {
          config: totalTeams && categories && rosterSlots
            ? { totalTeams, categories, rosterSlots }
            : null,
          confidence: totalTeams || categories || rosterSlots ? "low" : "failed",
          warnings,
        };
      }

      return {
        config: { totalTeams, categories, rosterSlots },
        confidence: "high",
        warnings,
      };
    }

    function detectTeamCount(): number | null {
      // Strategy 1: Count draft board columns (excluding player name column)
      const draftBoard = document.querySelector('[class*="draft-board"]');
      if (draftBoard) {
        const teamColumns = draftBoard.querySelectorAll('[class*="team-column"]');
        if (teamColumns.length > 0) {
          return teamColumns.length;
        }
      }

      // Strategy 2: Parse from league info header
      const leagueInfo = document.querySelector('[class*="league-info"]');
      if (leagueInfo) {
        const text = leagueInfo.textContent || "";
        const match = text.match(/(\d+)\s*teams?/i);
        if (match) {
          return parseInt(match[1], 10);
        }
      }

      // Strategy 3: Check URL parameters
      const url = new URL(window.location.href);
      const teams = url.searchParams.get("teams");
      if (teams) {
        return parseInt(teams, 10);
      }

      return null;
    }

    function detectCategories(): Category[] | null {
      // Strategy 1: Parse from draft board stat columns
      const statHeaders = document.querySelectorAll('[class*="stat-header"]');
      if (statHeaders.length > 0) {
        // Extract category names and map to Category objects
        // This is ESPN-specific and may need updates if ESPN changes DOM
        const categories: Category[] = [];
        statHeaders.forEach((header) => {
          const statName = header.textContent?.trim();
          if (statName) {
            categories.push(mapStatNameToCategory(statName));
          }
        });
        return categories;
      }

      // Strategy 2: Use standard 5x5 roto defaults as fallback
      return getDefault5x5Categories();
    }

    function detectRosterSlots(): Record<string, number> | null {
      // Strategy 1: Parse from league settings sidebar
      const settingsSidebar = document.querySelector('[class*="roster-settings"]');
      if (settingsSidebar) {
        const rosterLines = settingsSidebar.querySelectorAll('[class*="roster-slot"]');
        const slots: Record<string, number> = {};
        rosterLines.forEach((line) => {
          const text = line.textContent || "";
          const match = text.match(/^([A-Z0-9/]+):\s*(\d+)/);
          if (match) {
            slots[match[1]] = parseInt(match[2], 10);
          }
        });
        if (Object.keys(slots).length > 0) {
          return slots;
        }
      }

      // Strategy 2: Use standard 5x5 roster defaults
      return getDefault5x5Roster();
    }

    function mapStatNameToCategory(statName: string): Category {
      // Map ESPN stat display names to Category objects

      const categoryMap: Record<string, Category> = {
        "R": { category_id: "R", stat_key: "R", type: "counting", direction: "higher", name: "Runs" },
        "HR": { category_id: "HR", stat_key: "HR", type: "counting", direction: "higher", name: "Home Runs" },
        "RBI": { category_id: "RBI", stat_key: "RBI", type: "counting", direction: "higher", name: "RBI" },
        "SB": { category_id: "SB", stat_key: "SB", type: "counting", direction: "higher", name: "Stolen Bases" },
        "AVG": { category_id: "AVG", stat_key: "AVG", type: "rate", direction: "higher", name: "Batting Average" },
        "W": { category_id: "W", stat_key: "W", type: "counting", direction: "higher", name: "Wins" },
        "SV": { category_id: "SV", stat_key: "SV", type: "counting", direction: "higher", name: "Saves" },
        "K": { category_id: "K", stat_key: "K", type: "counting", direction: "higher", name: "Strikeouts" },
        "ERA": { category_id: "ERA", stat_key: "ERA", type: "rate", direction: "lower", name: "ERA" },
        "WHIP": { category_id: "WHIP", stat_key: "WHIP", type: "rate", direction: "lower", name: "WHIP" },
      };
      return categoryMap[statName] || {
        category_id: statName,
        stat_key: statName,
        type: "counting",
        direction: "higher",
        name: statName,
      };
    }
    ```

    ```typescript
    import { backOff } from 'exponential-backoff';

    export async function getRankingsWithRetry(
      config: DraftConfiguration,
    ): Promise<DraftRanking[]> {
      return backOff(
        () => getRankings(config),
        {
          numOfAttempts: 3,          // User constraint: 2-3 retries
          startingDelay: 1000,       // 1 second
          timeMultiple: 2,           // Exponential: 1s, 2s, 4s
          maxDelay: 5000,            // Cap at 5 seconds
          jitter: 'full',            // Randomize to prevent thundering herd
          retry: (error: any, attemptNumber: number) => {
            console.warn(`Retry attempt ${attemptNumber} after error:`, error);
            return true; // Always retry within attempt limit
          },
        }
      );
    }

    export async function getAuctionValuesWithRetry(
      config: DraftConfiguration,
    ): Promise<AuctionValue[]> {
      return backOff(
        () => getAuctionValues(config),
        {
          numOfAttempts: 3,
          startingDelay: 1000,
          timeMultiple: 2,
          maxDelay: 5000,
          jitter: 'full',
        }
      );
    }
    ```

    - Multiple selector strategies for robustness against ESPN changes
    - Warning banner shown when confidence is "low" or "failed"

    ```typescript
    // Near top of content.tsx, after imports
    let cachedLeagueConfig: DraftConfiguration | null = null;
    let configDetectionWarnings: string[] = [];

    async function refreshRankingsAndProjections() {
      const state = useDraftStore.getState();

      // Use cached config (don't re-detect per user constraint)
      if (!cachedLeagueConfig) {
        console.warn("No league config available for refresh");
        return;
      }

      try {
        // Fetch with retry logic
        const [rankings, auctionValues] = await Promise.all([
          getRankingsWithRetry(cachedLeagueConfig),
          getAuctionValuesWithRetry(cachedLeagueConfig),
        ]);

        // Update state
        state.setRankings(rankings);
        state.setAuctionValues(auctionValues);

        console.log("Rankings refreshed successfully", {
          rankingsCount: rankings.length,
          auctionValuesCount: auctionValues.length,
        });
      } catch (error) {
        // User constraint: keep stale data on failure after retries
        console.error("Failed to refresh rankings after retries, keeping stale data", error);
        // Don't throw - UI keeps showing stale rankings
      }
    }

    // In component initialization (useEffect or similar)
    useEffect(() => {
      // Detect config once at load
      const detection = detectLeagueConfig();

      if (detection.config) {
        cachedLeagueConfig = detection.config;
        useDraftStore.getState().initializeDraft({
          leagueId: 0, // Extension doesn't know league ID
          draftFormat: "snake", // Detect from page or default
          totalTeams: detection.config.totalTeams,
          rosterSize: Object.values(detection.config.rosterSlots).reduce((a, b) => a + b, 0),
          myTeamId: "1", // User must configure
        });
      } else {
        // Use default 5x5 config per user constraint
        cachedLeagueConfig = {
          totalTeams: 12,
          categories: getDefault5x5Categories(),
          rosterSlots: getDefault5x5Roster(),
        };
      }

      configDetectionWarnings = detection.warnings;

      // Initial rankings fetch
      refreshRankingsAndProjections();

      // Set up pick observer
      const cleanup = observeDraftBoard(async (pick) => {
        // Add pick to state
        useDraftStore.getState().addPick(pick);

        // Auto-refresh per user constraint
        await refreshRankingsAndProjections();
      });

      return cleanup;
    }, []);
    ```

    ```typescript
    export function observeDraftBoard(
      onPick: (pick: DraftPick) => void | Promise<void>,
    ): () => void {
      const observer = new MutationObserver((mutations) => {
        for (const mutation of mutations) {
          if (mutation.type === "childList") {
            mutation.addedNodes.forEach((node) => {
              const pickElements = getPickCandidatesFromNode(node);
              pickElements.forEach(async (pickElement) => {
                const pick = parsePickElement(pickElement);
                if (pick) {
                  await onPick(pick); // Support async handlers
                }
              });
            });
          }
        }
      });

      observer.observe(document.body, {
        childList: true,
        subtree: true,
      });

      return () => observer.disconnect();
    }
    ```
