import type { DraftPick, AuctionValue } from "@espn-fantasy/shared";
import type { InflationSignal } from "./types";

export function computeInflation(
  allPicks: DraftPick[],
  auctionValues: AuctionValue[],
  allTeamBudgets: Array<{ teamName: string; remainingBudget: number }>,
  totalTeams: number,
  myRemainingBudget: number,
): InflationSignal {
  // Build set of drafted player IDs
  const draftedIds = new Set(allPicks.map(p => p.playerId));

  // Remaining players with value > $1 (exclude min-bid filler players from denominator)
  const remainingPlayers = auctionValues.filter(
    av => !draftedIds.has(av.playerId) && av.dollarValue > 1
  );
  const totalRemainingValue = remainingPlayers.reduce(
    (sum, av) => sum + av.dollarValue, 0
  );

  // Total remaining budget across ALL teams
  // Prefer live DOM-scraped budgets if available (allTeamBudgets populated from pick train)
  let totalRemainingBudget: number;
  if (allTeamBudgets.length > 0) {
    totalRemainingBudget = allTeamBudgets.reduce(
      (sum, team) => sum + team.remainingBudget, 0
    );
  } else {
    // Fallback: estimate from picks
    const totalSpent = allPicks.reduce((sum, p) => sum + (p.cost ?? 0), 0);
    // Estimate initial budget: myRemainingBudget + my spend
    const mySpent = allPicks
      .filter(p => p.teamId === "myTeam" || p.cost !== undefined)
      .reduce((sum, p) => sum + (p.cost ?? 0), 0);
    const estimatedInitialBudget = myRemainingBudget + mySpent;
    const totalInitialBudget = estimatedInitialBudget > 0
      ? estimatedInitialBudget * totalTeams
      : 260 * totalTeams;
    totalRemainingBudget = Math.max(0, totalInitialBudget - totalSpent);
  }

  // Core formula from CONTEXT.md: inflation = remaining_budget / remaining_projected_value
  const rawFactor = totalRemainingValue > 0
    ? totalRemainingBudget / totalRemainingValue
    : 1.0;

  // Clamp per Phase 11 decision: 0.5 to 2.0
  const inflationFactor = Math.max(0.5, Math.min(2.0, rawFactor));

  // Low confidence when fewer than 20 valued players remain (late draft instability)
  const confidence = remainingPlayers.length >= 20 ? "high" : "low";

  return {
    inflationFactor,
    rawInflationFactor: rawFactor,
    totalRemainingBudget,
    totalRemainingValue,
    confidence,
    percentOverMarket: Math.round((inflationFactor - 1) * 100),
  };
}

import type { DraftPick, DraftRanking, DraftedPlayer } from "@espn-fantasy/shared";
import type { EnhancedScarcitySignal } from "./types";

const POSITION_TARGETS: Record<string, number> = {
  C: 1, "1B": 1, "2B": 1, "3B": 1, SS: 1, OF: 3, SP: 5, RP: 2,
};

function normalizePosition(pos: string): string | null {
  const upper = pos.toUpperCase();
  if (upper === "DH") return "UTIL";
  if (upper.includes("SP")) return "SP";
  if (upper.includes("RP")) return "RP";
  if (upper === "P") return "SP";
  if (upper.includes("OF")) return "OF";
  return POSITION_TARGETS[upper] ? upper : null;
}

export function computeEnhancedScarcity(
  rankings: DraftRanking[],
  allPicks: DraftPick[],
  teamRosters: Map<string, { teamId: string; teamName: string; playerIds: number[] }>,
  totalTeams: number,
  myRoster: DraftedPlayer[],
  myTeamId: string | null,
): EnhancedScarcitySignal[] {
  const draftedIds = new Set(allPicks.map(p => p.playerId));
  const signals: EnhancedScarcitySignal[] = [];

  // Sort rankings by VBD descending (don't trust incoming sort order -- per research pitfall #3)
  const sortedRankings = [...rankings].sort((a, b) => b.vbdScore - a.vbdScore);

  for (const [position, slotsPerTeam] of Object.entries(POSITION_TARGETS)) {
    if (position === "UTIL") continue;

    const topTierSize = totalTeams * slotsPerTeam;

    // All ranked players eligible for this position, sorted by VBD
    const eligibleRanked = sortedRankings.filter(r =>
      r.positions.some(p => normalizePosition(p) === position)
    );

    // Undrafted eligible players (preserving VBD sort)
    const eligibleUndrafted = eligibleRanked.filter(
      r => !draftedIds.has(r.playerId)
    );

    // Top-tier players still undrafted
    const topPlayers = eligibleRanked.slice(0, topTierSize);
    const supply = topPlayers.filter(p => !draftedIds.has(p.playerId)).length;
    const totalAvailable = eligibleUndrafted.length;

    // Demand: count teams that still need this position
    let demand = 0;
    const teamsAccountedFor = new Set<string>();

    for (const [teamId] of teamRosters) {
      teamsAccountedFor.add(teamId);
      const posCount = allPicks.filter(
        pick => pick.teamId === teamId &&
          pick.positions.some(p => normalizePosition(p) === position)
      ).length;
      if (posCount < slotsPerTeam) demand++;
    }
    // Unknown teams assumed to need all positions
    demand += Math.max(0, totalTeams - teamsAccountedFor.size);

    // Tier drop value: VBD gap between best available and first player in next tier
    let tierDropValue = 0;
    if (eligibleUndrafted.length >= 2) {
      const topAvailable = eligibleUndrafted[0];
      const nextTierPlayer = eligibleUndrafted.find(
        p => p.tier > topAvailable.tier
      );
      tierDropValue = nextTierPlayer
        ? topAvailable.vbdScore - nextTierPlayer.vbdScore
        : 0;
    }

    // Scarcity index: 0-1, combining demand/supply ratio (60%) and tier drop (40%)
    const demandSupplyRatio = supply > 0 ? demand / supply : (demand > 0 ? 2.0 : 0);
    const normalizedDemandSupply = Math.min(1.0, demandSupplyRatio / 2.0);
    const normalizedTierDrop = Math.min(1.0, tierDropValue / 50);
    const scarcityIndex = normalizedDemandSupply * 0.6 + normalizedTierDrop * 0.4;

    // Check if my team needs this position
    const myPosCount = myRoster.filter(player =>
      player.positions.some(p => normalizePosition(p) === position)
    ).length;
    const myTeamNeeds = myPosCount < slotsPerTeam;

    signals.push({
      position,
      supply,
      demand,
      totalAvailable,
      tierDropValue,
      scarcityIndex,
      myTeamNeeds,
    });
  }

  // Sort by scarcity index descending
  return signals.sort((a, b) => b.scarcityIndex - a.scarcityIndex);
}

import type { DraftedPlayer, DraftRanking, Category } from "@espn-fantasy/shared";
import { calculateCategorySGP, calculateFinalRate } from "@espn-fantasy/shared";
import type { CategoryGapSignal } from "./types";

export function computeCategoryGaps(
  myRoster: DraftedPlayer[],
  teamRosters: Map<string, { teamId: string; teamName: string; playerIds: number[] }>,
  rankingsMap: Map<number, DraftRanking>,
  categories: Category[],
): CategoryGapSignal[] {
  if (myRoster.length === 0 || categories.length === 0) return [];

  const signals: CategoryGapSignal[] = [];

  for (const cat of categories) {
    const statKey = cat.stat_key;
    const isRate = cat.type === "RATE";

    // My team's projected total for this category
    let myTotal: number;
    if (isRate && cat.formula_id) {
      // For rate stats: sum components across my roster, then compute rate
      const components: Record<string, number> = {};
      for (const player of myRoster) {
        const ranking = rankingsMap.get(player.playerId);
        const stats = ranking?.projectedStats ?? player.projectedStats ?? {};
        for (const [key, val] of Object.entries(stats)) {
          components[key] = (components[key] || 0) + (Number(val) || 0);
        }
      }
      try {
        myTotal = calculateFinalRate(cat.formula_id, {}, components);
      } catch {
        myTotal = 0;
      }
    } else {
      // For counting stats: direct sum
      myTotal = myRoster.reduce((sum, player) => {
        const ranking = rankingsMap.get(player.playerId);
        const stats = ranking?.projectedStats ?? player.projectedStats ?? {};
        return sum + (Number(stats[statKey]) || 0);
      }, 0);
    }

    // All teams' totals (for z-score comparison)
    const allTeamTotals: number[] = [];
    for (const [, roster] of teamRosters) {
      if (isRate && cat.formula_id) {
        const components: Record<string, number> = {};
        for (const pid of roster.playerIds) {
          const ranking = rankingsMap.get(pid);
          const stats = ranking?.projectedStats ?? {};
          for (const [key, val] of Object.entries(stats)) {
            components[key] = (components[key] || 0) + (Number(val) || 0);
          }
        }
        try {
          allTeamTotals.push(calculateFinalRate(cat.formula_id, {}, components));
        } catch {
          allTeamTotals.push(0);
        }
      } else {
        const teamTotal = roster.playerIds.reduce((sum, pid) => {
          const ranking = rankingsMap.get(pid);
          const stats = ranking?.projectedStats ?? {};
          return sum + (Number(stats[statKey]) || 0);
        }, 0);
        allTeamTotals.push(teamTotal);
      }
    }

    // Z-score: how far is my team from league average
    const gapZScore = allTeamTotals.length > 1
      ? calculateCategorySGP(myTotal, allTeamTotals)
      : 0;

    const leagueAvgTotal = allTeamTotals.length > 0
      ? allTeamTotals.reduce((a, b) => a + b, 0) / allTeamTotals.length
      : 0;

    // Risk score: for HIGHER_BETTER, negative z-score = risk. For LOWER_BETTER, positive z-score = risk (high ERA is bad).
    const adjustedZ = cat.direction === "LOWER_BETTER" ? -gapZScore : gapZScore;
    // Risk increases as z-score goes more negative (below average)
    const categoryRiskScore = Math.min(1.0, Math.max(0, (-adjustedZ) / 2.0));

    signals.push({
      category: cat.stat_key,
      myTeamTotal: Math.round(myTotal * 1000) / 1000, // 3 decimal places for rates
      leagueAvgTotal: Math.round(leagueAvgTotal * 1000) / 1000,
      gapZScore: Math.round(gapZScore * 100) / 100,
      categoryRiskScore: Math.round(categoryRiskScore * 100) / 100,
      direction: cat.direction as "HIGHER_BETTER" | "LOWER_BETTER",
    });
  }

  // Sort by risk score descending (most at-risk categories first)
  return signals.sort((a, b) => b.categoryRiskScore - a.categoryRiskScore);
}

import type { BudgetCurveSignal } from "./types";

export function computeBudgetCurve(
  remainingBudget: number,
  remainingRosterSpots: number,
  totalBudget: number,
  totalRosterSize: number,
  inflationFactor: number,
  currentPlayerDollarValue: number,
): BudgetCurveSignal {
  const minReserve = Math.max(0, remainingRosterSpots - 1); // $1 per remaining spot
  const usableBudget = Math.max(0, remainingBudget - minReserve);

  // Optimal spend curve: how much budget should remain at this draft stage
  const rosterProgress = totalRosterSize > 0
    ? 1 - (remainingRosterSpots / totalRosterSize)
    : 0;
  // Expect 90% of budget spent when roster is full (10% goes to $1 fillers)
  const expectedSpentFraction = rosterProgress * 0.9;
  const optimalRemainingSpend = totalBudget * (1 - expectedSpentFraction);

  // Budget pressure: >1.0 = have more than optimal (aggressive), <1.0 = behind (conserve)
  const budgetPressure = optimalRemainingSpend > 0
    ? remainingBudget / optimalRemainingSpend
    : 1.0;

  let pacingStatus: "ahead" | "on-track" | "behind";
  if (budgetPressure > 1.05) pacingStatus = "ahead";
  else if (budgetPressure < 0.95) pacingStatus = "behind";
  else pacingStatus = "on-track";

  // Max bid recommendation: inflation-adjusted value, capped by budget constraints
  const inflationAdjustedValue = Math.round(currentPlayerDollarValue * inflationFactor);
  const maxBidFromBudget = Math.floor(usableBudget * 0.35); // Never put >35% of usable on one player
  const maxBidFromValue = Math.round(inflationAdjustedValue * 1.10); // Allow 10% overpay on targets
  const maxBidRecommendation = Math.max(1, Math.min(maxBidFromValue, maxBidFromBudget, usableBudget));

  const avgPerPlayer = remainingRosterSpots > 0
    ? usableBudget / remainingRosterSpots
    : 0;

  return {
    remainingBudget,
    remainingRosterSpots,
    averageRemainingPerPlayer: Math.round(avgPerPlayer * 10) / 10,
    budgetPressure: Math.round(budgetPressure * 100) / 100,
    optimalRemainingSpend: Math.round(optimalRemainingSpend),
    maxBidRecommendation,
    pacingStatus,
    inflationAdjustedBid: Math.max(1, Math.min(inflationAdjustedValue, usableBudget)),
  };
}

import type {
  InflationSignal,
  EnhancedScarcitySignal,
  CategoryGapSignal,
  BudgetCurveSignal,
  DraftAlert,
} from "./types";

export function generateAlerts(
  inflation: InflationSignal,
  scarcity: EnhancedScarcitySignal[],
  categoryGaps: CategoryGapSignal[],
  budget: BudgetCurveSignal,
): DraftAlert[] {
  const alerts: DraftAlert[] = [];
  const now = Date.now();

  // Inflation alert: severity = (factor - 1) / 0.5 per research
  if (inflation.confidence === "high" && inflation.inflationFactor > 1.0) {
    const severity = Math.min(1.0, (inflation.inflationFactor - 1.0) / 0.5);
    if (severity > 0.65) {
      alerts.push({
        id: "inflation",
        type: "inflation",
        severity,
        title: `Market inflated ${inflation.percentOverMarket}%`,
        detail: `Players costing ${inflation.percentOverMarket}% above projected value. Adjust bids upward.`,
        timestamp: now,
      });
    }
  }

  // Scarcity alerts: severity = scarcityIndex (already 0-1)
  for (const s of scarcity) {
    if (s.scarcityIndex > 0.65) {
      alerts.push({
        id: `scarcity-${s.position}`,
        type: "scarcity",
        severity: s.scarcityIndex,
        title: `${s.position} scarcity critical`,
        detail: `${s.supply} elite left, ${s.demand} teams need. Drop: ${s.tierDropValue.toFixed(0)} VBD.`,
        position: s.position,
        timestamp: now,
      });
    }
  }

  // Category gap alerts: severity from categoryRiskScore (already 0-1)
  for (const g of categoryGaps) {
    if (g.categoryRiskScore > 0.65) {
      alerts.push({
        id: `cat-${g.category}`,
        type: "category_gap",
        severity: g.categoryRiskScore,
        title: `${g.category} deficit`,
        detail: `${Math.abs(g.gapZScore).toFixed(1)} std dev ${g.gapZScore < 0 ? "below" : "above"} league avg`,
        category: g.category,
        timestamp: now,
      });
    }
  }

  // Budget alert: severity = abs(1 - budgetPressure) clamped to 1.0
  const budgetSeverity = Math.min(1.0, Math.abs(1.0 - budget.budgetPressure));
  if (budgetSeverity > 0.65) {
    const direction = budget.budgetPressure > 1.0 ? "over" : "under";
    alerts.push({
      id: "budget",
      type: "budget",
      severity: budgetSeverity,
      title: `Budget ${budget.pacingStatus}`,
      detail: `${Math.round(Math.abs(1 - budget.budgetPressure) * 100)}% ${direction} optimal pace. Avg $${budget.averageRemainingPerPlayer}/player remaining.`,
      timestamp: now,
    });
  }

  // Tier break alerts: from scarcity signals where tier drop is large
  for (const s of scarcity) {
    const tierBreakSeverity = Math.min(1.0, s.tierDropValue / 30.0);
    if (tierBreakSeverity > 0.65 && s.myTeamNeeds) {
      alerts.push({
        id: `tier-break-${s.position}`,
        type: "tier_break",
        severity: tierBreakSeverity,
        title: `${s.position} tier break imminent`,
        detail: `${s.tierDropValue.toFixed(0)} VBD drop to next tier. Act now if you need ${s.position}.`,
        position: s.position,
        timestamp: now,
      });
    }
  }

  // Sort by severity descending
  return alerts.sort((a, b) => b.severity - a.severity);
}

import { computeInflation } from "./inflation";
import { computeEnhancedScarcity } from "./scarcity";
import { computeCategoryGaps } from "./category-gap";
import { computeBudgetCurve } from "./budget-curve";
import { generateAlerts } from "./alerts";

export { type DraftEngineSignals, type DraftEngineInput, type DraftAlert, type InflationSignal, type EnhancedScarcitySignal, type CategoryGapSignal, type BudgetCurveSignal } from "./types";

export function computeDraftEngine(input: DraftEngineInput): DraftEngineSignals {
  const t0 = performance.now();

  // Build rankingsMap for O(1) lookups
  const rankingsMap = new Map(input.rankings.map(r => [r.playerId, r]));

  // 1. Inflation model
  const inflation = computeInflation(
    input.allPicks,
    input.auctionValues,
    input.allTeamBudgets,
    input.totalTeams,
    input.remainingBudget,
  );

  // 2. Enhanced scarcity model
  const scarcity = computeEnhancedScarcity(
    input.rankings,
    input.allPicks,
    input.teamRosters,
    input.totalTeams,
    input.myRoster,
    input.myTeamId,
  );

  // 3. Category gap engine
  const categoryGaps = computeCategoryGaps(
    input.myRoster,
    input.teamRosters,
    rankingsMap,
    input.leagueCategories,
  );

  // 4. Budget curve model
  // Find current nominated player's dollar value
  const currentPlayerValue = input.currentAuctionPlayerId
    ? (input.auctionValues.find(av => av.playerId === input.currentAuctionPlayerId)?.dollarValue ?? 0)
    : 0;

  const remainingRosterSpots = Math.max(0, input.rosterSize - input.myRoster.length);

  // Compute initial budget from remaining + spent
  const mySpent = input.myRoster.reduce((sum, p) => sum + (p.cost ?? 0), 0);
  const totalBudget = input.remainingBudget + mySpent;

  const budgetCurve = computeBudgetCurve(
    input.remainingBudget,
    remainingRosterSpots,
    totalBudget,
    input.rosterSize,
    inflation.inflationFactor,
    currentPlayerValue,
  );

  // 5. Generate alerts from all signals
  const alerts = generateAlerts(inflation, scarcity, categoryGaps, budgetCurve);

  const elapsed = performance.now() - t0;
  if (elapsed > 300) {
    console.warn(`[DraftEngine] Recompute took ${elapsed.toFixed(1)}ms -- exceeds 300ms budget`);
  }

  return {
    inflation,
    scarcity,
    categoryGaps,
    budgetCurve,
    alerts,
    computeTimeMs: Math.round(elapsed * 10) / 10,
  };
}
