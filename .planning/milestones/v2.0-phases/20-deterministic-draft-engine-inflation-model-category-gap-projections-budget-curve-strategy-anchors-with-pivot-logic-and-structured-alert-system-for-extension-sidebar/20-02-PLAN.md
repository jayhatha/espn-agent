**packages/extension/lib/use-draft-engine.ts** -- React hook bridging Zustand to engine:

```typescript
import { useMemo } from "react";
import { useDraftStore } from "./draft-state";
import { computeDraftEngine } from "./engine";
import type { DraftEngineSignals } from "./engine";

export function useDraftEngineSignals(): DraftEngineSignals | null {
  const {
    rankings,
    auctionValues,
    allPicks,
    myRoster,
    remainingBudget,
    totalTeams,
    rosterSize,
    teamRosters,
    myTeamId,
    allTeamBudgets,
    leagueCategories,
    currentAuctionPlayer,
    draftFormat,
  } = useDraftStore();

  return useMemo(() => {
    // Only compute for auction drafts with data
    if (draftFormat !== "auction") return null;
    if (rankings.length === 0 && auctionValues.length === 0) return null;

    return computeDraftEngine({
      rankings,
      auctionValues,
      allPicks,
      myRoster,
      remainingBudget,
      totalTeams,
      rosterSize,
      teamRosters,
      myTeamId,
      allTeamBudgets,
      leagueCategories,
      currentAuctionPlayerId: currentAuctionPlayer?.playerId,
    });
  }, [
    // CRITICAL: Do NOT include currentAuctionPlayer in deps -- it changes every 200ms from
    // observeCurrentAuctionPlayer. Only recompute per-pick (allPicks change) or per-roster change.
    // The currentAuctionPlayer.playerId is read inside the useMemo but won't trigger recompute
    // on bid changes -- only on picks.
    rankings,
    auctionValues,
    allPicks,
    myRoster,
    remainingBudget,
    totalTeams,
    rosterSize,
    teamRosters.size, // Use .size instead of Map ref to avoid stale reference issues (Pitfall #6)
    myTeamId,
    allTeamBudgets,
    leagueCategories,
    draftFormat,
  ]);
}
```

NOTE about `currentAuctionPlayer`: Including it in the useMemo dependency array would cause recomputes every 200ms (from the MutationObserver). Per research Pitfall #5, exclude it. The `currentAuctionPlayerId` value will be stale between picks, but the engine's `maxBidRecommendation` updates correctly when `allPicks` changes (which happens after each completed auction). This is the correct tradeoff -- the nominated player's max bid updates once per completed pick, not once per bid increment.
