import { EventEmitter } from "events";

/**
 * In-process EventEmitter for fan-out of standings change events to SSE clients.
 * Single-process architecture — if multi-process scaling needed, swap for Redis pub/sub.
 * Singleton: imported by both the orchestrator (emit) and SSE route (subscribe).
 */
export const changeLogEmitter = new EventEmitter();
changeLogEmitter.setMaxListeners(100); // allow up to 100 concurrent SSE clients

export const CHANGE_LOG_EVENT = "standings_change";
```

// 8. Generate standings change log events (Phase 81)
// Non-blocking: errors here do not fail the snapshot
try {
  const bus = createNotificationBus(this.logger ?? pino());
  const changeLogAgent = new StandingsChangeLogAgent(this.db, this.logger ?? pino(), bus);
  const eventCount = await changeLogAgent.generateEvents(leagueId, snapshotDate);
  if (eventCount > 0) {
    // Signal SSE clients that new events are available
    changeLogEmitter.emit(CHANGE_LOG_EVENT, { leagueId, snapshotDate, eventCount });
  }
  this.logger?.info({ leagueId, snapshotDate, eventCount }, "Change log events generated");
} catch (err) {
  this.logger?.warn({ err, leagueId, snapshotDate }, "Change log generation failed — non-fatal");
}
```

**Key constraint:** Wrap in try/catch — change log failures must NOT propagate to fail the snapshot job.

```ts
// GET /api/leagues/:leagueId/standings/live-log
// SSE stream: emits historical events on connect, then live events as they arrive
app.get<{ Params: { leagueId: string }; Querystring: { limit?: string } }>(
  "/api/leagues/:leagueId/standings/live-log",
  {
    onRequest: [authenticate],
    preHandler: [requireLeagueAccess],
  },
  async (request, reply) => {
    const leagueId = parseIntParam(request.params.leagueId, "leagueId");
    const historyLimit = Math.min(parseInt(request.query.limit ?? "50", 10), 200);

    // Set SSE headers
    reply.raw.setHeader("Content-Type", "text/event-stream");
    reply.raw.setHeader("Cache-Control", "no-cache");
    reply.raw.setHeader("Connection", "keep-alive");
    reply.raw.setHeader("X-Accel-Buffering", "no"); // disable nginx buffering
    reply.raw.flushHeaders();

    function sendEvent(data: unknown): void {
      if (reply.raw.writable) {
        reply.raw.write(`data: ${JSON.stringify(data)}\n\n`);
      }
    }

    // 1. Send historical events immediately on connect
    try {
      const history = await db
        .selectFrom("standings_change_event")
        .selectAll()
        .where("league_id", "=", leagueId)
        .orderBy("occurred_at", "desc")
        .limit(historyLimit)
        .execute();

      // Send in chronological order (oldest first)
      for (const event of history.reverse()) {
        sendEvent({ type: "event", data: event });
      }
      sendEvent({ type: "history_end", count: history.length });
    } catch (err) {
      sendEvent({ type: "error", message: "Failed to load history" });
    }

    // 2. Subscribe to live events for this league
    function onNewEvent(payload: { leagueId: number; snapshotDate: string }) {
      if (payload.leagueId !== leagueId) return;
      // Fetch newly written events for this snapshot
      db.selectFrom("standings_change_event")
        .selectAll()
        .where("league_id", "=", leagueId)
        .where("snapshot_date", "=", payload.snapshotDate)
        .orderBy("occurred_at", "asc")
        .execute()
        .then((events) => {
          for (const event of events) {
            sendEvent({ type: "event", data: event });
          }
        })
        .catch(() => {/* ignore — client will receive next heartbeat */});
    }

    changeLogEmitter.on(CHANGE_LOG_EVENT, onNewEvent);

    // 3. Heartbeat every 30s to keep connection alive
    const heartbeat = setInterval(() => {
      sendEvent({ type: "heartbeat", ts: Date.now() });
    }, 30_000);

    // 4. Cleanup on disconnect
    request.raw.on("close", () => {
      clearInterval(heartbeat);
      changeLogEmitter.off(CHANGE_LOG_EVENT, onNewEvent);
      reply.raw.end();
    });

    // Return a never-resolving promise so Fastify doesn't close the response
    await new Promise<void>((resolve) => {
      request.raw.on("close", resolve);
    });
  },
);

// GET /api/leagues/:leagueId/standings/change-log
// REST endpoint: paginated history (for initial load without SSE)
app.get<{ Params: { leagueId: string }; Querystring: { limit?: string; before?: string } }>(
  "/api/leagues/:leagueId/standings/change-log",
  {
    onRequest: [authenticate],
    preHandler: [requireLeagueAccess],
  },
  async (request, reply) => {
    try {
      const leagueId = parseIntParam(request.params.leagueId, "leagueId");
      const limit = Math.min(parseInt(request.query.limit ?? "50", 10), 200);

      let query = db
        .selectFrom("standings_change_event")
        .selectAll()
        .where("league_id", "=", leagueId);

      if (request.query.before) {
        query = query.where("occurred_at", "<", new Date(request.query.before));
      }

      const events = await query
        .orderBy("occurred_at", "desc")
        .limit(limit)
        .execute();

      return { events };
    } catch (err) {
      return reply.code(400).send({ error: err instanceof Error ? err.message : "Invalid request" });
    }
  },
);
```
