### Task 2: NotificationBus interface + adapters

**New file:** `packages/backend/src/utils/notification-bus.ts`

```ts
import type { StandingsChangeEventRow } from "../db/schema.js";
import type { BaseLogger } from "pino";

export interface NotificationBus {
  emit(event: Omit<StandingsChangeEventRow, "id" | "occurred_at">): Promise<void>;
}

/**
 * No-op bus — used when SLACK_WEBHOOK_URL is unset.
 * Zero runtime cost, zero external calls.
 */
export class NoOpNotificationBus implements NotificationBus {
  async emit(_event: Omit<StandingsChangeEventRow, "id" | "occurred_at">): Promise<void> {
    // intentionally empty
  }
}

/**
 * Slack incoming webhook bus.
 * Activated when SLACK_WEBHOOK_URL env var is set.
 * Formats a human-readable Slack message per event.
 */
export class SlackWebhookNotificationBus implements NotificationBus {
  constructor(
    private readonly webhookUrl: string,
    private readonly logger: BaseLogger,
  ) {}

  async emit(event: Omit<StandingsChangeEventRow, "id" | "occurred_at">): Promise<void> {
    const sign = event.direction === "gained" ? "📈" : "📉";
    const deltaStr = event.roto_delta > 0
      ? `+${event.roto_delta.toFixed(1)}`
      : event.roto_delta.toFixed(1);

    let text = `${sign} *${event.category_name}* ${deltaStr} pts`;
    if (event.passed_team_name) {
      text += event.direction === "gained"
        ? ` — passed *${event.passed_team_name}*`
        : ` — passed by *${event.passed_team_name}*`;
    }
    if (event.proximate_cause) {
      text += `\n> ${event.proximate_cause.description}`;
    }

    try {
      const res = await fetch(this.webhookUrl, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ text }),
      });
      if (!res.ok) {
        this.logger.warn({ status: res.status }, "Slack webhook delivery failed");
      }
    } catch (err) {
      this.logger.warn({ err }, "Slack webhook emit error — continuing");
    }
  }
}

/**
 * Factory: returns appropriate bus based on env.
 * Call once at app startup.
 */
export function createNotificationBus(logger: BaseLogger): NotificationBus {
  const webhookUrl = process.env.SLACK_WEBHOOK_URL;
  if (webhookUrl) {
    logger.info("NotificationBus: Slack webhook adapter active");
    return new SlackWebhookNotificationBus(webhookUrl, logger);
  }
  logger.info("NotificationBus: no-op adapter (set SLACK_WEBHOOK_URL to enable Slack)");
  return new NoOpNotificationBus();
}
```

### Task 3: StandingsChangeLogAgent

**New file:** `packages/backend/src/agents/standings-change-log.ts`

The agent:
1. Finds the two most recent snapshots for the league
2. For each team, runs the same attribution math as Phase 80 (reuse `computeActualPerformanceDelta` etc.)
3. For each category where `|roto_delta| >= 0.5`:
   - Determines direction (gained/lost)
   - Finds passed team: the team whose rank we crossed
   - Formats proximate_cause from `topDriverPlayer` or `topCompetitorMoved`
4. Inserts rows into `standings_change_event`
5. Emits each event via `NotificationBus`
6. Returns count of events written

Key implementation points:
- Guard: if fewer than 2 snapshots exist, return early (no diff possible)
- Guard: skip writing duplicate events for same (league_id, team_id, snapshot_date, category_id) — use `ON CONFLICT DO NOTHING` with unique constraint
- Add unique constraint: `UNIQUE (league_id, team_id, snapshot_date, category_id)` to migration
- Noise gate: only emit/persist when `Math.abs(rotoDelta) >= 0.5`
- Passed team logic: after ranking at `currProjected`, find the team immediately above (if we gained) or immediately below (if we lost) whose rank changed relative to prevProjected ranking
- Proximate cause:
  - `actual_performance` bucket: `{ playerName: topDriverPlayer, stat: category_name, value: |actual delta|, description: "player drove category change" }`
  - `projection_change` bucket: `{ playerName: topProjectionDriver, stat: "RoS Projection", value: projDelta, description: "projection updated" }`
  - `competitor_movement` bucket: `{ playerName: topCompetitorMoved, stat: "competitor stats", value: 0, description: "competitor team moved" }`
