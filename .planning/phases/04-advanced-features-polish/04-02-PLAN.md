  opponentId: number;
  isHome: boolean;
  venue: string;
  gameDate: string;
  gamePk: number;

export class MLBStatsAPIAdapter {
  private client: AxiosInstance;

  constructor(private logger?: Logger) {
    this.client = axios.create({

      timeout: 10000,
    });
  }

**LineupOptimizerAgent** (`packages/backend/src/agents/lineup-optimizer.ts`):

```typescript
export interface LineupRecommendation {
  slotId: string;
  slotName: string;
  currentPlayerId: number;
  currentPlayerName: string;
  recommendedPlayerId: number;
  recommendedPlayerName: string;
  reason: string;
  projectedValueGain: number; // SGP gain from the swap
  currentPlayerValue: number;
  recommendedPlayerValue: number;
}

export interface LineupOptimizationResult {
  recommendations: LineupRecommendation[];
  totalProjectedGain: number;
  snapshotDate: string;
  isOptimal: boolean; // true if no swaps recommended
}

export class LineupOptimizerAgent {
  constructor(
    private db: Kysely<Database>,
    private mlbAdapter: MLBStatsAPIAdapter,
    private logger?: Logger,
  ) {}

  async getRecommendations(
    leagueId: number,
    teamId: number,
    snapshotDate: string,
  ): Promise<LineupOptimizationResult>
}
```

Implementation:
1. Load league settings from `league_settings_version` (categories, roster_slots).
2. Load team roster from `team_roster_snapshot` for snapshotDate. This gives player IDs and their current slot assignments (active/bench/IL).
3. Load player projections from `player_projection` for all rostered players (join on player_id, use latest fresh projections_run).
4. Load today's MLB schedule via `this.mlbAdapter.getSchedule(snapshotDate, snapshotDate)`. Build a Set of MLB team abbreviations that have games today.
5. For each player, calculate a "today value" score:
   - If player's team has NO game today: value = 0 (can't contribute)
   - If player's team has a game: value = sum of category SGPs from their RoS projections / remaining games estimate. This is a simplified per-game value.
6. Apply greedy lineup optimization (from research):
   - Sort active roster slots by restrictiveness (fewer eligible positions first: C, 1B, SS before UTIL)
   - Sort players by today value descending
   - For each slot, assign best eligible unassigned player
7. Compare optimized lineup vs current lineup. For each slot where the recommended player differs from the current player, create a `LineupRecommendation`.
8. Return result with isOptimal = true if no swaps needed.

**Greedy algorithm detail:** The greedy approach sorts slots by restrictiveness (fewest eligible positions first) so that catcher (only eligible at C) gets filled before UTIL (eligible for all). This prevents a good catcher from being "wasted" in UTIL while a worse player sits at C.

**StreamingPitcherAgent** (`packages/backend/src/agents/streaming-pitcher.ts`):

```typescript
export interface StreamingCandidate {
  playerId: number;
  playerName: string;
  mlbTeam: string;
  opponent: string;
  isHome: boolean;
  venue: string;
  gameDate: string;
  compositeScore: number;
  projectedStats: StatComponents; // RoS projection scaled to single start
  weekStarts: number; // Number of starts this week (2-start pitchers bonus)
  ownership: number | null; // ESPN ownership % if available
  scoreBreakdown: {
    projectionScore: number;
    opponentScore: number;
    multiStartBonus: number;
  };
}

export interface StreamingResult {
  candidates: StreamingCandidate[];
  dateRange: { start: string; end: string };
  totalCandidates: number;
}

export class StreamingPitcherAgent {
  constructor(
    private db: Kysely<Database>,
    private mlbAdapter: MLBStatsAPIAdapter,
    private logger?: Logger,
  ) {}

  async getRecommendations(
    leagueId: number,
    teamId: number,
    options?: {
      startDate?: string; // Default: today
      lookAheadDays?: number; // Default: 7
      limit?: number; // Default: 20
    }
  ): Promise<StreamingResult>
}
```

Implementation:
1. Load league settings (categories — need to know which pitching cats matter: K, W, SV, ERA, WHIP).
2. Determine date range: startDate to startDate + lookAheadDays.
3. Get probable pitchers via `this.mlbAdapter.getProbablePitchers()` for each day in range.
4. Get multi-start map via `this.mlbAdapter.getMultiStartPitchers()` for the week.
5. Load all rostered player IDs across all teams from `team_roster_snapshot` for the league (to exclude already-rostered pitchers).
6. Load pitcher projections from `player_projection` (where IP > 0 in stat_components_json). Join with `player` table for name and mlb_team.
7. Match probable pitchers to projections using player name fuzzy matching (or exact match on name). Only keep pitchers NOT on any team's roster (free agents).
8. Score each candidate:
   - **Projection score (0-50):** Scale from RoS projections. Per-start K projection * 2 (max ~20pts). Per-start ERA bonus: (5.0 - projected_ERA) * 5 if under 5.0. Per-start WHIP bonus: (2.0 - projected_WHIP) * 10 if under 2.0.
   - **Opponent score (0-30):** Use league-wide team batting rank as proxy. Since we don't have opponent stats in DB, use a simpler heuristic: if probable pitcher is announced, give base 15 points. Home pitchers get +5 bonus.
   - **Multi-start bonus (0-20):** If pitcher's team has 2+ starts in the week window, +15 points. If pitcher specifically is a probable for 2 games in the range, +20.
9. Sort by compositeScore descending. Return top `limit` candidates.

Log: date range, total probable pitchers found, matched to projections count, final candidate count.

**Import patterns:** Use `.js` extensions for local imports. Import types from `../../../shared/dist/index.js`. Import Database from `../db/schema.js`. Import MLBStatsAPIAdapter from `../adapters/mlb-statsapi.js`.
