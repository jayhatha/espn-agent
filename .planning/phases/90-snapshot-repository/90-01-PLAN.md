```typescript
import type { Kysely } from "kysely";
import type { Database } from "./schema.js";

export class SnapshotRepository {
  constructor(private db: Kysely<Database>) {}

  /** Roster rows for a league+date, joined with player. Optional teamId filter. */
  getRosterForSnapshot(leagueId: number, snapshotDate: string, teamId?: number) {
    let q = this.db
      .selectFrom("team_roster_snapshot as roster")
      .innerJoin("player", "roster.player_id", "player.id")
      .select([
        "roster.team_id",
        "roster.player_id",
        "roster.roster_status",
        "roster.roster_slot",
        "player.name as player_name",
        "player.mlb_team",
        "player.positions",
        "player.espn_asv",
      ])
      .where("roster.league_id", "=", leagueId)
      .where("roster.snapshot_date", "=", snapshotDate);

    if (teamId !== undefined) {
      q = q.where("roster.team_id", "=", teamId);
    }

    return q.execute();
  }

  /** All category totals for a league on a given snapshot date. */
  getLeagueCategoryTotals(leagueId: number, snapshotDate: string) {
    return this.db
      .selectFrom("team_daily_category_totals")
      .selectAll()
      .where("league_id", "=", leagueId)
      .where("snapshot_date", "=", snapshotDate)
      .execute();
  }

  /**
   * Player projections for a projections run.
   * If playerIds is provided and non-empty, filters to those players only.
   * If playerIds is [] (empty), skips the IN clause — Kysely generates invalid
   * SQL for WHERE player_id IN (), so we return all rows for the run instead.
   */
  getPlayerProjections(projectionsRunId: number, playerIds?: number[]) {
    let q = this.db
      .selectFrom("player_projection")
      .select(["player_id", "stat_components_json"])
      .where("projections_run_id", "=", projectionsRunId);

    if (playerIds !== undefined && playerIds.length > 0) {
      q = q.where("player_id", "in", playerIds);
    }

    return q.execute();
  }

  /**
   * League settings joined through the snapshot for a given date.
   * Matches the join pattern used by trade-recommendation.ts callers.
   */
  getLeagueSettings(leagueId: number, snapshotDate: string) {
    return this.db
      .selectFrom("league_daily_snapshot")
      .innerJoin(
        "league_settings_version",
        "league_daily_snapshot.settings_version_id",
        "league_settings_version.id",
      )
      .select(["league_settings_version.settings_json"])
      .where("league_daily_snapshot.league_id", "=", leagueId)
      .where("league_daily_snapshot.snapshot_date", "=", snapshotDate)
      .executeTakeFirst();
  }

  /** Most recent snapshot for a league by date. */
  getLatestSnapshot(leagueId: number) {
    return this.db
      .selectFrom("league_daily_snapshot")
      .selectAll()
      .where("league_id", "=", leagueId)
      .orderBy("snapshot_date", "desc")
      .executeTakeFirst();
  }
}
```
