  <files>packages/extension/lib/engine/nomination-strategy.ts</files>
  <action>
Create `packages/extension/lib/engine/nomination-strategy.ts` with a single export:

`recommendNomination(availablePlayers: PoolPlayer[], sgpValues: number[], myTeamState: SimTeamState, opponentModel: OpponentModel, opponentStates: SimTeamState[], topN: number = 5): NominationRecommendation[]`

**Algorithm (heuristic, not simulation-based — per research open question #2):**

For each available player, compute a trapScore (0-1):

1. **Budget drain factor** (0-0.4): `min(0.4, player.dollars / 50)`. Expensive players drain more opponent budget. Cap at 0.4 to prevent dominating the score.

2. **Opponent need factor** (0-0.3): Count how many opponents have remaining slots at this player's positions. `opponentNeedCount = sum over opponentStates of (any of player.positions has slotsRemaining > 0 for this opponent ? 1 : 0)`. Normalize: `min(0.3, 0.3 * opponentNeedCount / max(1, opponentStates.length))`. More opponents needing this position = higher bidding war probability.

3. **My non-need factor** (0-0.3): I want to nominate players I DON'T need. `myNeedScore = any of player.positions has myTeamState.slotsRemaining > 0 ? 1.0 : 0.0`. Non-need factor = `0.3 * (1 - myNeedScore)`. Players at positions I've already filled are better trap nominations.

4. **trapScore = budgetDrainFactor + opponentNeedFactor + myNonNeedFactor**

5. **Reason generation:** Build a human-readable string:
   - If budgetDrainFactor > 0.2: "Expensive ($XX) — drains opponent budget"
   - If opponentNeedFactor > 0.15: "N opponents need this position"
   - If myNonNeedFactor > 0.15: "You don't need this position"
   - Combine with " + " separator

6. **expectedDrain = player.dollars * opponentNeedFactor / 0.3** (rough estimate of how much opponent will spend)

7. Sort by trapScore descending, return top N (default 5).

**Edge cases:**
- If opponentStates is empty, opponentNeedFactor = 0 for all players.
- If no available players, return empty array.
- Floor trapScore at 0.01 (never return exactly 0).

**Pure function requirements:** No Zustand, no DOM access. Accepts all data as parameters.

**File header comment:** Same style as other engine modules — describe purpose, exports, pure function guarantee.

if (newSimCacheKey === simulationCacheKey && cachedSimulation !== null) {

  input.leagueCategories.length > 0 &&
  input.teamRosters.size >= 3 &&
  input.myTeamId != null

  try {
    const simT0 = performance.now();

    // Build simulation draft state from current engine state
    const simState = buildSimDraftState(
      allPlayers, // From computeRosterOptimization scope — need to hoist
      input,
      opponentModel, // From computeRosterOptimization scope — need to hoist
      activeSlots,
      remainingSlots,
    );

    if (simState !== null) {
      const SIMS_PER_DECISION = 200;
      const TEMPERATURE = 1.0;

      // Baseline win probability (no decision scenario)
      // Run sims on current state to get expected finish
      const baselineSimResults: number[][] = [];
      for (let s = 0; s < SIMS_PER_DECISION; s++) {
        const completed = simulateFullDraft(
          allPlayers, sgpValues, input.leagueCategories, simState, TEMPERATURE,
        );
        baselineSimResults.push(
          computeSimStandings(completed.teamStates, input.leagueCategories),
        );
      }
      const winProbability = aggregateOutcomes(
        simState.myTeamIndex, baselineSimResults, simState.teamStates.length,
      );

      // Decision comparison (bid vs pass) when a player is nominated
      let currentDecision: DecisionComparison | null = null;
      if (input.currentAuctionPlayerId != null && optimization.maxBid > 0) {
        // Find nominated player index in allPlayers
        const nominated = input.auctionValues.find(
          av => av.playerId === input.currentAuctionPlayerId,
        );
        if (nominated) {
          const nName = normalizeName(nominated.playerName ?? '');
          const nomIdx = allPlayers.findIndex(p => p.normalizedName === nName);
          if (nomIdx >= 0) {
            const candidateActions: CandidateAction[] = [
              { type: "bid", label: `Bid $${optimization.maxBid}`, playerIndex: nomIdx, bidPrice: optimization.maxBid },
              { type: "pass", label: "Pass", playerIndex: nomIdx },
            ];
            currentDecision = compareDecisions(
              candidateActions, simState, allPlayers, sgpValues,
              input.leagueCategories, SIMS_PER_DECISION, TEMPERATURE,
            );
          }
        }
      }

      // Nomination strategy
      let nominationStrategy: NominationRecommendation[] | null = null;
      const myIdx = simState.myTeamIndex;
      const oppStates = simState.teamStates.filter((_, i) => i !== myIdx);
      if (oppStates.length > 0) {
        const availableForNom = Array.from(simState.availableIndices)
          .map(i => allPlayers[i])
          .filter(Boolean);

          availableForNom, availableSgps,
          simState.teamStates[myIdx], opponentModel, oppStates, 5,

      const simElapsed = performance.now() - simT0;
      simulation = {
        currentDecision,
        winProbability,
        nominationStrategy,
        computeTimeMs: Math.round(simElapsed * 10) / 10,
      };
    }

    cachedSimulation = simulation;
    simulationCacheKey = newSimCacheKey;
  } catch (e) {
    console.warn("[DraftEngine] Forward simulation failed:", e);
