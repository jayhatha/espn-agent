    ```typescript
    import { describe, it, expect, beforeEach, vi, afterEach } from "vitest";
    import { ESPNAdapter } from "../espn.js";
    import { createMockLogger } from "../../__tests__/test-helpers.js";

    describe("ESPNAdapter", () => {
      let adapter: ESPNAdapter;

      beforeEach(() => {
        adapter = new ESPNAdapter({ logger: createMockLogger() });
        vi.useFakeTimers();
      });

      afterEach(() => {
        vi.restoreAllMocks();
        vi.useRealTimers();
      });

      describe("rate limiting (DEBT-09)", () => {
        it("should throttle consecutive calls with 2-second delay", async () => {
          const mockAxios = vi.spyOn(adapter["axios"], "get").mockResolvedValue({
            data: { id: 123 },
            status: 200,
            statusText: "OK",
            headers: {},
            config: {} as any
          });

          // First call - no delay
          const promise1 = adapter.fetchLeague(123, 2026, {});
          await vi.advanceTimersByTimeAsync(0);

          // Second call - should wait 2s
          const promise2 = adapter.fetchLeague(456, 2026, {});

          // Should not call axios yet (waiting for throttle)
          expect(mockAxios).toHaveBeenCalledTimes(1);

          // Advance 2 seconds
          await vi.advanceTimersByTimeAsync(2000);

          // Now second call should execute
          await promise1;
          await promise2;

          expect(mockAxios).toHaveBeenCalledTimes(2);
        });

        it("should handle 429 errors with Retry-After header", async () => {
          const error = {
            response: {
              status: 429,
              headers: { "retry-after": "30" }
            }
          };

          vi.spyOn(adapter["axios"], "get").mockRejectedValueOnce(error);

          await expect(adapter.fetchLeague(123, 2026, {})).rejects.toMatchObject({
            response: { status: 429 }
          });
        });
      });

      describe("cache bounds (DEBT-09)", () => {
        it("should use LRU cache with max size", () => {
          // Access private cache to verify it's LRUCache instance
          const cache = adapter["cache"];

          expect(cache).toBeDefined();
          expect(cache.max).toBe(500); // From Plan 13-05
        });

        it("should evict old entries when max size reached", async () => {
          const cache = adapter["cache"];

          // Fill cache beyond max
          for (let i = 0; i < 600; i++) {
            cache.set(`key-${i}`, { data: i });
          }

          // Size should be at most 500
          expect(cache.size).toBeLessThanOrEqual(500);

          // Oldest entries should be evicted
          expect(cache.has("key-0")).toBe(false);
          expect(cache.has("key-599")).toBe(true);
        });
      });
    });
    ```

    ```typescript
    import { describe, it, expect } from "vitest";
    import { SignupSchema, LoginSchema, parseIntParam } from "../../utils/validation.js";

    describe("Auth validation (DEBT-03)", () => {
      describe("SignupSchema", () => {
        it("should validate correct signup data", () => {
          const valid = {
            email: "test@example.com",
            password: "password123",
            username: "testuser"
          };

          const result = SignupSchema.safeParse(valid);

          expect(result.success).toBe(true);
          if (result.success) {
            expect(result.data.email).toBe("test@example.com");
          }
        });

        it("should reject invalid email", () => {
          const invalid = {
            email: "not-an-email",
            password: "password123"
          };

          const result = SignupSchema.safeParse(invalid);

          expect(result.success).toBe(false);
          if (!result.success) {
            expect(result.error.errors[0].path).toContain("email");
          }
        });

        it("should reject short password", () => {
          const invalid = {
            email: "test@example.com",
            password: "short"
          };

          const result = SignupSchema.safeParse(invalid);

          expect(result.success).toBe(false);
          if (!result.success) {
            expect(result.error.errors[0].path).toContain("password");
          }
        });

        it("should handle missing fields", () => {
          const invalid = { email: "test@example.com" };

          const result = SignupSchema.safeParse(invalid);

          expect(result.success).toBe(false);
        });
      });

      describe("LoginSchema", () => {
        it("should validate correct login data", () => {
          const valid = {
            email: "test@example.com",
            password: "anypassword"
          };

          const result = LoginSchema.safeParse(valid);

          expect(result.success).toBe(true);
        });

        it("should reject invalid email", () => {
          const invalid = {
            email: "bad-email",
            password: "password"
          };

          const result = LoginSchema.safeParse(invalid);

          expect(result.success).toBe(false);
        });
      });

      describe("parseIntParam", () => {
        it("should parse valid integer", () => {
          const result = parseIntParam("123", "leagueId");

          expect(result).toBe(123);
        });

        it("should throw on invalid integer", () => {
          expect(() => parseIntParam("abc", "leagueId")).toThrow("Invalid leagueId");
        });

        it("should throw on NaN", () => {
          expect(() => parseIntParam("", "teamId")).toThrow("Invalid teamId");
        });
      });
    });
    ```
