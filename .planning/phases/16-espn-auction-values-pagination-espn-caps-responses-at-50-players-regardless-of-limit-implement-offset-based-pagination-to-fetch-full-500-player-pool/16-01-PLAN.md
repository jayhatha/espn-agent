<objective>
Fix ESPN returning only 50 players in auction value and free agent fetches by moving x-fantasy-filter from URL params to HTTP headers, and add offset-based pagination to fetchPreDraftAuctionValues to retrieve the full 500+ player pool.

Purpose: ESPN ignores x-fantasy-filter when sent as a URL query parameter; it only reads this filter from HTTP request headers. This causes ESPN to use its 50-player default cap regardless of the limit specified in the filter JSON. Fixing the transport layer and adding pagination will give the auction validation endpoint and draft assistant access to the full player pool.

Output: Modified espn.ts with both methods corrected, pagination loop in fetchPreDraftAuctionValues, and sortDraftRanks added to auction filter.
</objective>

   ```typescript
   let offset = 0;
   const allPlayers: any[] = [];

   for (let i = 0; i < MAX_ITERATIONS; i++) {
     const headers: Record<string, string> = {};
     if (auth?.SWID && auth?.espn_s2) {
       headers.Cookie = `SWID=${auth.SWID}; espn_s2=${auth.espn_s2}`;
     }

     // KEY FIX: x-fantasy-filter sent as HTTP header, not URL param
     // Add sortDraftRanks to surface players with auction values first
     headers["x-fantasy-filter"] = JSON.stringify({
       players: {
         filterStatus: { value: ["FREEAGENT", "WAIVERS", "ONTEAM"] },
         filterSlotIds: { value: ALL_POSITION_SLOT_IDS },
         sortDraftRanks: { sortPriority: 100, sortAsc: true, value: "STANDARD" },
         limit: batchSize,
         offset,
       },
     });

     const params: Record<string, unknown> = {
       view: ["kona_player_info"],
       scoringPeriodId: 1,
     };

     this.logger?.info(
       { leagueId, season, offset, batchSize },
       "Fetching ESPN pre-draft auction values batch",
     );

     await throttleEspnCall();

     let response;
     try {
       response = await this.client.get(url, { params, headers });
     } catch (error: any) {
       if (error.response?.status === 429) {
         const retryAfter = error.response.headers['retry-after'] || 60;
         this.logger?.warn({ retryAfter }, "ESPN rate limit hit - backing off");
       }
       throw error;
     }

     if (response.status !== 200) {
       throw new Error(`ESPN API returned status ${response.status}`);
     }

     const validatedData = validateEspnResponse(
       ESPNFreeAgentsResponseSchema,
       response.data,
       `fetchPreDraftAuctionValues(${leagueId}, ${season}, offset=${offset})`,
       this.logger
     );

     const batch = (validatedData?.players as any[]) || [];

     this.logger?.info({
       leagueId,
       season,
       offset,
       batchSize,
       received: batch.length,
       totalSoFar: allPlayers.length + batch.length,
     }, "ESPN auction values batch fetched");

     allPlayers.push(...batch);

     // Partial page signals end of results
     if (batch.length < batchSize) break;

     offset += batchSize;
   }
   ```
