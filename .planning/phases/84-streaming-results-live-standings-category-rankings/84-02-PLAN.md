```typescript
    api.getChangeLogHistory(leagueIdNum, 50, "live") as Promise<{

    }>,

```

Update `api.getChangeLogHistory` signature in `lib/api.ts` to accept optional `source` param:

```typescript
async getChangeLogHistory(
  leagueId: number,
  limit = 50,
  source?: string,
): Promise<{ events: ChangeEvent[] }> {
  const params = new URLSearchParams({ limit: String(limit) });
  if (source) params.set("source", source);
  return this.get(`/api/leagues/${leagueId}/standings/change-log?${params}`);
}
```

### Task 4: Add stat-type chip to EventCard

Derive a compact stat abbreviation from `category_name`:

```typescript
/** Maps verbose category names to compact abbreviations for the chip. */
function statAbbrev(categoryName: string): string {
  const name = categoryName.toUpperCase();
  if (name.includes("HOME RUN") || name === "HR") return "HR";
  if (name.includes("STOLEN") || name === "SB") return "SB";
  if (name.includes("STRIKEOUT") || name === "K") return "K";
  if (name === "ERA" || name.includes("EARN")) return "ERA";
  if (name === "WHIP" || name.includes("WALKS+HITS")) return "WHIP";
  if (name.includes("AVG") || name.includes("BATTING")) return "AVG";
  if (name.includes("OPS")) return "OPS";
  if (name.includes("OBP")) return "OBP";
  if (name.includes("RBI")) return "RBI";
  if (name.includes("RUN") && !name.includes("HOME")) return "R";
  if (name.includes("WIN") || name === "W") return "W";
  if (name.includes("SAVE") || name === "SV") return "SV";
  if (name.includes("HOLD") || name === "HLD") return "HLD";
  // fallback: first 3-5 chars of category name
  return categoryName.slice(0, 4).toUpperCase();
}
```

Add a compact stat chip inside `EventCard`, between the category badge and the narrative:

```tsx
{/* Stat abbreviation chip */}
<span className="inline-flex items-center px-1 py-0.5 rounded text-[10px] font-mono font-medium bg-surface-2 text-content-tertiary mr-1">
  {statAbbrev(event.category_name)}
</span>
```
