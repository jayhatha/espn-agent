    // Add to DraftRankingsAgent class

    /**
     * Calculate projected final season standings for a draft roster
     * Used by extension to show impact of draft picks on projected finish
     */
    async calculateProjectedStandings(
      config: DraftConfiguration,
      rosters: TeamRoster[],  // Each team's drafted players
      options?: {
        projectionsRunId?: number;
      },
    ): Promise<ProjectedStandingsResult> {
      this.logger?.info({ numTeams: rosters.length }, "Calculating projected standings");

      // 1. For each team, aggregate projected stats from drafted players
      const teamTotals: TeamProjectedStats[] = [];

      for (const roster of rosters) {
        const playerIds = roster.players.map(p => p.playerId);

        // Fetch projections for this team's players
        let query = this.db
          .selectFrom("player_projection")
          .innerJoin("player", "player.espn_player_id", "player_projection.player_id")
          .select([
            "player.id",
            "player_projection.projected_stats",
          ])
          .where("player.id", "in", playerIds);

        if (options?.projectionsRunId) {
          query = query.where("player_projection.run_id", "=", options.projectionsRunId);
        } else {
          // Use latest projections
          const latestRun = await this.db
            .selectFrom("projections_run")
            .select("id")
            .where("status", "=", "fresh")
            .orderBy("uploaded_at", "desc")
            .executeTakeFirst();

          if (!latestRun) {
            throw new Error("No fresh projections available");
          }

          query = query.where("player_projection.run_id", "=", latestRun.id);
        }

        const projections = await query.execute();

        // Aggregate stats using component math (from shared/src/formulas.ts)
        const aggregated = this.aggregateTeamStats(
          projections.map(p => p.projected_stats),
          config.categories,
        );

        teamTotals.push({
          teamId: roster.teamId,
          teamName: roster.teamName,
          stats: aggregated,
        });
      }

      // 2. Rank teams per category
      const standings = this.rankTeams(teamTotals, config.categories, config.totalTeams);

      return {
        standings,
        generatedAt: new Date().toISOString(),
      };
    }

    private aggregateTeamStats(
      playerStats: StatComponents[],
      categories: Category[],
    ): StatComponents {
      // Use component-based aggregation (never average rates)
      // Pattern from StandingsProjectionAgent
      const components: StatComponents = {};

      for (const player of playerStats) {
        for (const [key, value] of Object.entries(player)) {
          components[key] = (components[key] || 0) + value;
        }
      }

      return components;
    }

    private rankTeams(
      teamTotals: TeamProjectedStats[],
      categories: Category[],
      totalTeams: number,
    ): ProjectedOverallRow[] {
      const standings: ProjectedOverallRow[] = [];

      // For each category, rank teams
      const categoryRanks: Record<string, Record<number, number>> = {};

      for (const category of categories) {
        const sorted = [...teamTotals].sort((a, b) => {
          const aValue = this.getCategoryValue(a.stats, category);
          const bValue = this.getCategoryValue(b.stats, category);

          return category.direction === "higher"
            ? bValue - aValue
            : aValue - bValue;
        });

        categoryRanks[category.category_id] = {};
        sorted.forEach((team, index) => {
          categoryRanks[category.category_id][team.teamId] = index + 1;
        });
      }

      // Calculate roto points per team
      for (const team of teamTotals) {
        let totalPoints = 0;

        for (const category of categories) {
          const rank = categoryRanks[category.category_id][team.teamId];
          const points = totalTeams - rank + 1;  // Roto scoring
          totalPoints += points;
        }

        standings.push({
          team_id: team.teamId,
          teamName: team.teamName,
          projected_overall_points: totalPoints,
          projected_overall_rank: 0,  // Assigned after sorting
        });
      }

      // Sort by total points, assign overall rank
      standings.sort((a, b) => b.projected_overall_points - a.projected_overall_points);
      standings.forEach((team, index) => {
        team.projected_overall_rank = index + 1;
      });

      return standings;
    }

    private getCategoryValue(stats: StatComponents, category: Category): number {
      if (category.type === "rate") {
        // Calculate rate from components
        const formula = getRateFormula(category.stat_key);
        return formula(stats);
      } else {
        return stats[category.stat_key] || 0;
      }
    }

    // Types
    interface TeamRoster {
      teamId: number;
      teamName: string;
      players: { playerId: number }[];
    }

    interface TeamProjectedStats {
      teamId: number;
      teamName: string;
      stats: StatComponents;
    }

    interface ProjectedStandingsResult {
      standings: ProjectedOverallRow[];
      generatedAt: string;
    }
    ```

    **Key implementation notes:**
    - Reuses component-based aggregation from StandingsProjectionAgent
    - Uses latest fresh projections if projectionsRunId not specified
    - Roto scoring: points = numTeams - rank + 1 per category
    - Rate stats calculated from aggregated components (never averaged)
    - Returns standings sorted by total roto points

    // Add to draftRoutes function in draft.ts

    // POST /api/draft/projected-standings - League-agnostic projected standings
    app.post<{
      Body: {
        config: DraftConfiguration;
        rosters: TeamRoster[];
        projectionsRunId?: number;
      };
    }>(
      "/api/draft/projected-standings",
      {
        config: {
          rateLimit: {
            max: 50,
            timeWindow: "1 minute",
          },
        },
      },
      async (request, reply) => {
        const { config, rosters, projectionsRunId } = request.body;

        // Validate config
        if (!config || !config.categories || !config.rosterSlots || !config.totalTeams) {
          return reply.code(400).send({ error: "Invalid draft configuration" });
        }

        // Validate rosters
        if (!Array.isArray(rosters) || rosters.length === 0) {
          return reply.code(400).send({ error: "Rosters array required" });
        }

        for (const roster of rosters) {
          if (!roster.teamId || !roster.players || !Array.isArray(roster.players)) {
            return reply.code(400).send({ error: "Invalid roster format" });
          }
        }

        try {
          const draftAgent = new DraftRankingsAgent(db, request.log as any);

          const result = await draftAgent.calculateProjectedStandings(
            config,
            rosters,
            { projectionsRunId },
          );

          return {
            standings: result.standings,
            meta: {
              totalTeams: config.totalTeams,
              categories: config.categories.map(c => c.name),
              generatedAt: result.generatedAt,
            },
          };
        } catch (error) {
          request.log.error({ error }, "Failed to calculate projected standings");
          return reply.code(500).send({
            error: "Failed to calculate projected standings",
            details: error instanceof Error ? error.message : String(error),
          });
        }
      },
    );
    ```

    **Endpoint design decisions (per user constraint - Claude's discretion):**
    - POST with config in body (not query params) - avoids URL length limits, cleaner for complex objects
    - No authentication required (generic endpoint, no league-specific data)
    - Rate limit: 50 requests/minute (reasonable for draft frequency)
    - Returns full standings array + metadata
