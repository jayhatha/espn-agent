```typescript
interface SimulationConfig {
  numSimulations: number;      // default 200
  totalTeams: number;          // 13
  budget: number;              // 260
  myTeamIndex: number;         // 0
  myTemperature: number;       // 0.15 (near-greedy value strategy)
  opponentTemperature: number; // 1.5 (market follower: moderate randomness)
  rosterSlots: Record<string, number>;  // C:1,1B:1,2B:1,SS:1,3B:1,MI:1,CI:1,OF:4,UTIL:3,P:9,BN:5
  categories: Category[];      // 10 standard categories
}

interface SingleSimResult {
  myRoster: PoolPlayer[];       // Players drafted by our team
  totalFgValue: number;         // Sum of FG dollar values of our roster
  totalPricePaid: number;       // Sum of what we paid
  profit: number;               // totalFgValue - totalPricePaid
  rotoPoints: number;           // Our roto point total
  overallRank: number;          // 1-based rank among all teams (1 = best)
  categoryFinishes: Record<string, number>;  // stat_key -> 1-based rank
  top4Categories: number;       // Count of categories where rank <= 4
  positionSpending: Record<string, number>;  // position group -> total $ spent
}

interface AggregatedResults {
  results: SingleSimResult[];
  stats: {
    value: { mean: number; median: number; p10: number; p90: number };
    profit: { mean: number; median: number; p10: number; p90: number };
    rotoPoints: { mean: number; median: number; p10: number; p90: number };
    top4Rate: number;           // fraction of sims with >= 6 top-4 finishes
    avgTop4Count: number;       // average number of top-4 category finishes
  };
  categoryFinishDistribution: Record<string, number[]>;  // stat_key -> array of 13 counts
  playerFrequency: Array<{ name: string; count: number; pct: number }>;
}
```

**Functions:**

1. `buildSimPlayerPool()`: Import the four FG JSON files directly (using `../../assets/fangraphs-*.json` relative path since this file is in `lib/engine/`). Call `buildFgPlayerPool()` from archetype.ts with an empty draftedNames set. Merge undraftedHitters and undraftedPitchers into a single PoolPlayer[]. Return the combined pool.

2. `getDefaultConfig()`: Returns a SimulationConfig with locked constants:
   - 13 teams, $260 budget, **28 slots: C:1, 1B:1, 2B:1, SS:1, 3B:1, MI:1, CI:1, OF:4, UTIL:3, P:9, BN:5** (CONTEXT.md says 28 slots with BN:5)
   - 10 categories: R, RBI, HR, SB, OPS (COUNTING/HIGHER_BETTER), K, W, SV (COUNTING/HIGHER_BETTER), ERA (RATE/LOWER_BETTER/formula_id:era), WHIP (RATE/LOWER_BETTER/formula_id:whip)
   - OPS is treated as COUNTING for SGP/standings (consistent with simulation-strategy.test.ts)
   - myTemperature: 0.15, opponentTemperature: 1.5, 200 sims

3. `runSingleAuctionSim(pool, sgpValues, config): SingleSimResult`:

   **Auction loop** (simplified auction with price discovery):
   - Initialize SimDraftState: 13 teams, each with budget 260, slotsRemaining from config.rosterSlots
   - **Maintain a parallel `rosterIndices: number[][]` array** (one sub-array per team). On each player assignment, push the playerIdx onto the winning team's rosterIndices sub-array. After simulation, use `rosterIndices[config.myTeamIndex]` to reconstruct our roster.
   - Round-robin nomination: each team nominates a player
     - Our team (idx 0): nominate player with worst value ratio (high dollars vs SGP) to drain opponents
     - Opponents: random pick from top 15 available by SGP
   - Bidding: each team with a valid slot for the nominated player submits a max bid
     - Our team: `min(playerFgDollars, budget - (slotsRemaining - 1))` — bid full FG value, never overpay
     - Opponents: `floor(playerFgDollars * (1 + gaussianNoise(0, 0.25)))`, clamped to `[1, budget - (slotsRemaining - 1)]`
     - Only teams with a valid slot for the player participate
   - Winner = highest bidder (ties: first in order). Winner pays their bid. Player assigned to winner via `draftPlayerToTeam`.
   - Continue until all teams full or pool empty.

   **Draft composition enforcement for our team (idx 0):**
   - Target: 7 SP + 5 RP = 12 pitchers total (9 P slots + 3 bench pitchers), 2 bench hitters
   - Track SP/RP counts for our team. When our team has 7 SP, skip SP nominations and deprioritize SP in bidding. Same for 5 RP cap.
   - For bench slots: first 3 bench picks should be pitchers (check if we still need pitchers), last 2 should be hitters.

   **overallRank computation:**
   - After computing roto points for all teams via `computeSimStandings`, sort all teams' roto points descending.
   - Our team's overallRank = 1-based position in that sorted order. If tied, use lower rank (worse).

   **Result computation:**
   - Reconstruct our roster: `rosterIndices[config.myTeamIndex].map(idx => pool[idx])`
   - totalFgValue = sum of player.dollars for our roster
   - totalPricePaid = sum of prices actually paid by our team (tracked during auction loop)
   - profit = totalFgValue - totalPricePaid
   - categoryFinishes: use `computeCategoryRanks(teamStates, categories)` — implement this helper that returns per-category 1-based ranks for each team
   - top4Categories = count of categories where our rank <= 4
   - positionSpending: group our roster by position (hitter/pitcher) and sum costs

   gaussianRandom(): Box-Muller: `Math.sqrt(-2 * Math.log(Math.random())) * Math.cos(2 * Math.PI * Math.random())`

4. `runSimulations(config?: Partial<SimulationConfig>): AggregatedResults`:
   - Merge config with getDefaultConfig()
   - Build pool via buildSimPlayerPool()
   - Build SGP vectors via buildSGPVectors()
   - Run config.numSimulations iterations of runSingleAuctionSim()
   - Aggregate: mean/median/p10/p90 for value, profit, rotoPoints; top4Rate; categoryFinishDistribution; playerFrequency (top 30)
   - Helper: `percentile(arr, p)` — sort, return value at index floor(arr.length * p)
