```typescript
"use client";

import { GLOSSARY } from "@espn-fantasy/shared";
import { Tooltip } from "./Tooltip";

interface TermProps {
  id: keyof typeof GLOSSARY;
  children?: React.ReactNode;
}

export function Term({ id, children }: TermProps) {
  const definition = GLOSSARY[id];
  if (!definition) {
    return <span>{children ?? id}</span>;
  }
  return <Tooltip content={definition}>{children ?? id.toUpperCase()}</Tooltip>;
}
```

1. Add a grouping utility function (can be defined at the top of the file or inline):

```typescript
function groupByTopCategory(
  recs: WaiverRecommendation[]
): { category: string; recs: WaiverRecommendation[] }[] {
  const groups: Record<string, WaiverRecommendation[]> = {};

  for (const rec of recs) {
    // Find the category with highest absolute deltaPoints
    const topCat = rec.categoryImpacts
      ?.filter((ci) => Math.abs(ci.deltaPoints) > 0.3)
      .sort((a, b) => Math.abs(b.deltaPoints) - Math.abs(a.deltaPoints))[0];

    const groupKey = topCat ? topCat.categoryName : "Best Overall";
    if (!groups[groupKey]) groups[groupKey] = [];
    groups[groupKey].push(rec);
  }

  // Sort groups: specific categories first (sorted by average deltaPoints descending), "Best Overall" last
  return Object.entries(groups)
    .map(([category, recs]) => ({ category, recs }))
    .sort((a, b) => {
      if (a.category === "Best Overall") return 1;
      if (b.category === "Best Overall") return -1;
      const avgA = a.recs.reduce((sum, r) => {
        const top = r.categoryImpacts?.sort((x, y) => Math.abs(y.deltaPoints) - Math.abs(x.deltaPoints))[0];
        return sum + (top?.deltaPoints ?? 0);
      }, 0) / a.recs.length;
      const avgB = b.recs.reduce((sum, r) => {
        const top = r.categoryImpacts?.sort((x, y) => Math.abs(y.deltaPoints) - Math.abs(x.deltaPoints))[0];
        return sum + (top?.deltaPoints ?? 0);
      }, 0) / b.recs.length;
      return avgB - avgA;
    });
}
```
