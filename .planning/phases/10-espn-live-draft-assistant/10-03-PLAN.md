```typescript
import type { DraftPick } from '@espn-fantasy/shared';

/**
 * Observes ESPN draft board for new picks
 * Returns cleanup function to disconnect observer
 */
export function observeDraftBoard(onPick: (pick: DraftPick) => void): () => void {
  // ESPN draft board structure (adjust based on actual DOM inspection)
  // NOTE: These selectors are placeholders - must be verified against real ESPN draft page
  const draftBoardSelector = '.draft-board, .draftboard-container, [class*="draftBoard"]';
  const pickRowSelector = '.pick-row, .draft-pick, [class*="pickRow"]';

  const observer = new MutationObserver((mutations) => {
    for (const mutation of mutations) {
      if (mutation.type === 'childList') {
        // Check for new pick rows
        mutation.addedNodes.forEach((node) => {
          if (node instanceof HTMLElement) {
            // Check if node itself is a pick row or contains one
            const pickElement = node.matches(pickRowSelector)
              ? node
              : node.querySelector(pickRowSelector);

            if (pickElement) {
              const pick = parsePickElement(pickElement);
              if (pick) {
                console.log('[ESPN Draft Assistant] Detected pick:', pick);
                onPick(pick);
              }
            }
          }
        });
      }
    }
  });

  // Find draft board and start observing
  const draftBoard = document.querySelector(draftBoardSelector);
  if (draftBoard) {
    console.log('[ESPN Draft Assistant] Observing draft board:', draftBoard);
    observer.observe(draftBoard, {
      childList: true,
      subtree: true,
    });
  } else {
    console.warn('[ESPN Draft Assistant] Draft board not found. Selectors may need updating.');
    // Retry after delay (board might load asynchronously)
    setTimeout(() => {
      const retryBoard = document.querySelector(draftBoardSelector);
      if (retryBoard) {
        console.log('[ESPN Draft Assistant] Draft board found on retry');
        observer.observe(retryBoard, { childList: true, subtree: true });
      }
    }, 2000);
  }

  // Return cleanup function
  return () => observer.disconnect();
}

/**
 * Extract pick data from ESPN DOM element
 * NOTE: Selectors are placeholders - must be verified against real ESPN draft page
 */
export function parsePickElement(pickElement: Element): DraftPick | null {
  try {
    // ESPN-specific DOM parsing (requires manual inspection)
    const playerNameEl = pickElement.querySelector('.player-name, [class*="playerName"]');
    const teamEl = pickElement.querySelector('.team-abbrev, [class*="teamAbbrev"]');
    const positionEl = pickElement.querySelector('.player-position, [class*="position"]');

    const playerName = playerNameEl?.textContent?.trim();
    const teamName = teamEl?.textContent?.trim();
    const positions = positionEl?.textContent?.trim().split(',').map(p => p.trim()) || [];

    if (!playerName || !teamName) {
      console.warn('[ESPN Draft Assistant] Incomplete pick data:', { playerName, teamName });
      return null;
    }

    // Extract IDs and pick number from data attributes (if available)
    const playerId = parseInt(pickElement.getAttribute('data-player-id') || '0', 10);
    const pickNumber = parseInt(pickElement.getAttribute('data-pick-number') || '0', 10);
    const teamId = pickElement.getAttribute('data-team-id') || teamName;

    // Auction cost (if present)
    const costEl = pickElement.querySelector('.pick-cost, [class*="cost"]');
    const cost = costEl ? parseInt(costEl.textContent?.replace(/\D/g, '') || '0', 10) : undefined;

    return {
      playerId: playerId || Date.now(), // Fallback to timestamp if no ID
      playerName,
      teamId,
      teamName,
      pickNumber: pickNumber || 0,
      timestamp: Date.now(),
      cost,
      positions,
    };
  } catch (error) {
    console.error('[ESPN Draft Assistant] Error parsing pick element:', error);
    return null;
  }
}

/**
 * Parse league ID from ESPN draft page URL
 * Example: https://fantasy.espn.com/baseball/draft?leagueId=123456
 */
export function parseLeagueIdFromUrl(): number | null {
  const urlParams = new URLSearchParams(window.location.search);
  const leagueId = urlParams.get('leagueId');
  return leagueId ? parseInt(leagueId, 10) : null;
}

/**
 * Detect draft format from ESPN page
 * Heuristic: look for auction-specific UI elements or URL params
 */
export function detectDraftFormat(): 'snake' | 'auction' {
  // Check URL for format param
  const urlParams = new URLSearchParams(window.location.search);
  const format = urlParams.get('draftType') || urlParams.get('format');
  if (format?.toLowerCase().includes('auction')) {
    return 'auction';
  }

  // Check for auction-specific elements
  const auctionIndicators = document.querySelectorAll('[class*="auction"], [class*="bid"]');
  if (auctionIndicators.length > 0) {
    return 'auction';
  }

  // Default to snake
  return 'snake';
}
```

```typescript
import { defineContentScript } from 'wxt/sandbox';
import { createRoot } from 'react-dom/client';
import { observeDraftBoard, parseLeagueIdFromUrl, detectDraftFormat } from '../lib/dom-observers';
import { useDraftStore } from '../lib/draft-state';
import DraftSidebar from '../components/DraftSidebar';
import contentStyles from './content.css?inline';

export default defineContentScript({
  matches: ['https://fantasy.espn.com/baseball/draft*'],
  runAt: 'document_idle',

  async main() {
    console.log('[ESPN Draft Assistant] Content script loaded');

    // Parse draft metadata from page
    const leagueId = parseLeagueIdFromUrl();
    const draftFormat = detectDraftFormat();

    if (!leagueId) {
      console.warn('[ESPN Draft Assistant] Could not parse league ID from URL');
      return;
    }

    console.log('[ESPN Draft Assistant] Draft detected:', { leagueId, draftFormat });

    // Initialize draft state if not already set
    const state = useDraftStore.getState();
    if (!state.leagueId) {
      // TODO: Get these from backend API or ESPN page
      state.initializeDraft({
        leagueId,
        draftFormat,
        totalTeams: 12, // Default, should be parsed from ESPN page
        rosterSize: 25, // Default, should be from league settings
        myTeamId: 'user', // TODO: Detect user's team from ESPN page
        budget: 260, // Auction only
      });
    }

    // Create sidebar container
    const sidebarRoot = document.createElement('div');
    sidebarRoot.id = 'espn-draft-assistant-root';
    sidebarRoot.style.cssText = `
      position: fixed;
      top: 0;
      right: 0;
      width: 400px;
      height: 100vh;
      z-index: 999999;
      box-shadow: -2px 0 8px rgba(0,0,0,0.1);
    `;

    // Attach Shadow DOM for style isolation
    const shadowRoot = sidebarRoot.attachShadow({ mode: 'open' });

    // Inject Tailwind styles into Shadow DOM
    const styleSheet = document.createElement('style');
    styleSheet.textContent = contentStyles;
    shadowRoot.appendChild(styleSheet);

    // Mount React app into Shadow DOM
    const reactRoot = document.createElement('div');
    shadowRoot.appendChild(reactRoot);

    document.body.appendChild(sidebarRoot);

    // Render sidebar
    const root = createRoot(reactRoot);
    root.render(<DraftSidebar />);

    console.log('[ESPN Draft Assistant] Sidebar rendered');

    // Start observing draft board for picks
    const cleanup = observeDraftBoard((pick) => {
      useDraftStore.getState().addPick(pick);

      // If pick is for user's team, add to roster
      if (pick.teamId === state.myTeamId) {
        // TODO: Fetch player projections to populate DraftedPlayer
        const draftedPlayer = {
          playerId: pick.playerId,
          playerName: pick.playerName,
          positions: pick.positions,
          pickNumber: pick.pickNumber,
          cost: pick.cost,
          projectedStats: {}, // TODO: Load from rankings
        };
        useDraftStore.getState().updateRoster(draftedPlayer);
      }
    });

    // Cleanup on page unload
    window.addEventListener('beforeunload', cleanup);
  },
});
```

```typescript
import React from 'react';
import { useDraftStore } from '../lib/draft-state';

export default function DraftSidebar() {
  const {
    sidebarVisible,
    toggleSidebar,
    leagueId,
    draftFormat,
    currentPick,
    myRoster,
    remainingBudget,
  } = useDraftStore();

  if (!sidebarVisible) {
    return (
      <button
        onClick={toggleSidebar}
        className="fixed top-4 right-4 bg-blue-600 text-white px-3 py-2 rounded-lg shadow-lg hover:bg-blue-700"
      >
        Show Assistant
      </button>
    );
  }

  return (
    <div className="h-full bg-white flex flex-col">
      {/* Header */}
      <div className="bg-blue-600 text-white p-4 flex justify-between items-center">
        <div>
          <h1 className="text-lg font-bold">ESPN Draft Assistant</h1>
          <p className="text-sm opacity-90">
            League {leagueId} • {draftFormat === 'snake' ? 'Snake' : 'Auction'} Draft
          </p>
        </div>
        <button
          onClick={toggleSidebar}
          className="text-white hover:bg-blue-700 px-2 py-1 rounded"
        >
          ✕
        </button>
      </div>

      {/* Draft Status */}
      <div className="bg-gray-100 p-4 border-b">
        <div className="flex justify-between text-sm">
          <span>Pick #{currentPick}</span>
          {draftFormat === 'auction' && (
            <span className="font-semibold">${remainingBudget} remaining</span>
          )}
        </div>
        <div className="mt-2 text-sm text-gray-600">
          Roster: {myRoster.length} players
        </div>
      </div>

      {/* Main content area - will be populated by subsequent plans */}
      <div className="flex-1 overflow-y-auto p-4">
        <p className="text-gray-500 text-center mt-8">
          Player recommendations will appear here
        </p>
      </div>
    </div>
  );
}
```
