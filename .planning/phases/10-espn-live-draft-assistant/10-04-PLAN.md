**2. Create DraftRecommendations component (components/DraftRecommendations.tsx):**

```typescript
import React, { useMemo } from 'react';
import { useDraftStore } from '../lib/draft-state';
import PlayerCard from './PlayerCard';

export default function DraftRecommendations() {
  const {
    rankings,
    selectedPosition,
    showOnlyNeeds,
    setPositionFilter,
    toggleNeedsFilter,
    myRoster,
    allPicks,
  } = useDraftStore();

  // Filter out already-drafted players
  const draftedPlayerIds = new Set(allPicks.map(p => p.playerId));
  const availablePlayers = useMemo(() => {
    return rankings.filter(p => !draftedPlayerIds.has(p.playerId));
  }, [rankings, allPicks]);

  // Apply position filter
  const filteredPlayers = useMemo(() => {
    let players = availablePlayers;

    if (selectedPosition) {
      players = players.filter(p => p.positions.includes(selectedPosition));
    }

    if (showOnlyNeeds) {
      // Simplified needs logic: positions with < 2 players on roster
      const positionCounts: Record<string, number> = {};
      myRoster.forEach(p => {
        p.positions.forEach(pos => {
          positionCounts[pos] = (positionCounts[pos] || 0) + 1;
        });
      });

      const neededPositions = Object.entries(positionCounts)
        .filter(([, count]) => count < 2)
        .map(([pos]) => pos);

      players = players.filter(p =>
        p.positions.some(pos => neededPositions.includes(pos))
      );
    }

    return players;
  }, [availablePlayers, selectedPosition, showOnlyNeeds, myRoster]);

  const positions = ['C', '1B', '2B', '3B', 'SS', 'OF', 'SP', 'RP'];

  return (
    <div className="flex flex-col h-full">
      {/* Filter controls */}
      <div className="p-3 bg-gray-50 border-b space-y-2">
        <div className="flex gap-1 flex-wrap">
          <button
            onClick={() => setPositionFilter(null)}
            className={`
              px-2 py-1 text-xs rounded
              ${!selectedPosition ? 'bg-blue-600 text-white' : 'bg-white text-gray-700 border'}
            `}
          >
            All
          </button>
          {positions.map(pos => (
            <button
              key={pos}
              onClick={() => setPositionFilter(pos)}
              className={`
                px-2 py-1 text-xs rounded
                ${selectedPosition === pos ? 'bg-blue-600 text-white' : 'bg-white text-gray-700 border'}
              `}
            >
              {pos}
            </button>
          ))}
        </div>

        <label className="flex items-center text-sm text-gray-700 cursor-pointer">
          <input
            type="checkbox"
            checked={showOnlyNeeds}
            onChange={toggleNeedsFilter}
            className="mr-2"
          />
          Show only positional needs
        </label>
      </div>

      {/* Player list */}
      <div className="flex-1 overflow-y-auto">
        {filteredPlayers.length === 0 ? (
          <div className="p-4 text-center text-gray-500">
            {rankings.length === 0 ? 'Loading rankings...' : 'No players match filters'}
          </div>
        ) : (
          filteredPlayers.slice(0, 50).map((player, index) => (
            <PlayerCard
              key={player.playerId}
              player={player}
              rank={index + 1}
            />
          ))
        )}
      </div>
    </div>
  );
}
```

**1. Create SnakeTurnPairs component (components/SnakeTurnPairs.tsx):**

```typescript
import React, { useMemo } from 'react';
import { useDraftStore } from '../lib/draft-state';
import { getSnakeTurnPairs } from '../lib/value-calculations';
import type { PairRecommendation } from '../lib/value-calculations';

export default function SnakeTurnPairs() {
  const { rankings, myRoster, allPicks, totalTeams, currentPick } = useDraftStore();

  // Determine if user is at the turn
  const isAtTurn = useMemo(() => {
    // In snake draft, turn picks are at positions totalTeams and totalTeams+1
    // Example: 12-team draft, turn at picks 12-13, 24-25, 36-37, etc.
    const pickMod = currentPick % (totalTeams * 2);
    return pickMod === totalTeams || pickMod === totalTeams + 1 || pickMod === 0;
  }, [currentPick, totalTeams]);

  // Filter out already-drafted players
  const draftedPlayerIds = new Set(allPicks.map(p => p.playerId));
  const availablePlayers = useMemo(() => {
    return rankings.filter(p => !draftedPlayerIds.has(p.playerId));
  }, [rankings, allPicks]);

  // Calculate pair recommendations
  const pairs = useMemo(() => {
    if (!isAtTurn || availablePlayers.length < 2) return [];

    // Simplified: pass empty categories array (full implementation would fetch from league settings)
    return getSnakeTurnPairs(availablePlayers, myRoster, []);
  }, [isAtTurn, availablePlayers, myRoster]);

  if (!isAtTurn) {
    return null; // Only show at turn
  }

  return (
    <div className="p-3 bg-yellow-50 border-b border-yellow-200">
      <h3 className="font-semibold text-sm mb-2 text-yellow-900">
        🎯 Turn Pair Recommendations
      </h3>
      <p className="text-xs text-yellow-700 mb-3">
        You have back-to-back picks. Consider these pairs:
      </p>

      <div className="space-y-2">
        {pairs.slice(0, 3).map((pair, index) => (
          <div key={index} className="bg-white p-2 rounded border border-yellow-300">
            <div className="flex justify-between items-start mb-1">
              <div className="text-sm font-medium text-gray-900">
                {pair.player1.playerName} + {pair.player2.playerName}
              </div>
              <div className="text-xs text-gray-500">
                {pair.combinedValue.toFixed(1)} VBD
              </div>
            </div>
            <div className="text-xs text-gray-600">
              {pair.rationale}
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
```

**2. Create RosterSummary component (components/RosterSummary.tsx):**

```typescript
import React, { useMemo } from 'react';
import { useDraftStore } from '../lib/draft-state';

export default function RosterSummary() {
  const { myRoster, draftFormat, remainingBudget } = useDraftStore();

  // Calculate position counts
  const positionCounts = useMemo(() => {
    const counts: Record<string, number> = {};
    myRoster.forEach(player => {
      player.positions.forEach(pos => {
        counts[pos] = (counts[pos] || 0) + 1;
      });
    });
    return counts;
  }, [myRoster]);

  // Calculate category totals (simplified)
  const categoryTotals = useMemo(() => {
    const totals: Record<string, number> = {
      HR: 0,
      RBI: 0,
      SB: 0,
      W: 0,
      SV: 0,
    };

    myRoster.forEach(player => {
      Object.keys(totals).forEach(stat => {
        if (player.projectedStats[stat]) {
          totals[stat] += player.projectedStats[stat] as number;
        }
      });
    });

    return totals;
  }, [myRoster]);

  return (
    <div className="p-3 bg-gray-50 border-b">
      <h3 className="font-semibold text-sm mb-2">Your Roster ({myRoster.length})</h3>

      {/* Position breakdown */}
      <div className="grid grid-cols-4 gap-2 text-xs mb-3">
        {Object.entries(positionCounts).map(([pos, count]) => (
          <div key={pos} className="bg-white p-2 rounded border text-center">
            <div className="font-semibold text-gray-900">{pos}</div>
            <div className="text-gray-600">{count}</div>
          </div>
        ))}
      </div>

      {/* Category totals */}
      <div className="bg-white p-2 rounded border text-xs">
        <div className="font-semibold text-gray-700 mb-1">Projected Totals</div>
        <div className="grid grid-cols-3 gap-2">
          {Object.entries(categoryTotals).map(([stat, total]) => (
            <div key={stat}>
              <span className="text-gray-600">{stat}:</span>{' '}
              <span className="font-semibold">{Math.round(total)}</span>
            </div>
          ))}
        </div>
      </div>

      {draftFormat === 'auction' && (
        <div className="mt-2 bg-blue-50 p-2 rounded border border-blue-200 text-sm text-center">
          <span className="font-semibold text-blue-900">${remainingBudget}</span>
          <span className="text-blue-700"> remaining</span>
        </div>
      )}
    </div>
  );
}
```
