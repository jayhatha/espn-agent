### Task 2: Create LivePollAgent

```typescript
import type { Kysely } from "kysely";
import type { BaseLogger } from "pino";
import pino from "pino";
import type { Database } from "../db/schema.js";
import { ESPNAdapter } from "../adapters/espn.js";
import { StandingsProjectionAgent } from "./standings-projection.js";
import { StandingsChangeLogAgent } from "./standings-change-log.js";
import { createNotificationBus } from "../utils/notification-bus.js";
import {
  changeLogEmitter,
  CHANGE_LOG_EVENT,
  LIVE_POLL_EVENT,
} from "../utils/change-log-emitter.js";
import { assignRanksWithTies } from "../utils/roto-ranking.js";
import { getEspnAuthFromEnv } from "../utils/espn-env.js";

/**
 * LivePollAgent
 * Fetches ESPN stats on a 15-minute cadence during game hours,
 * updates today's category/rank/overall tables, recomputes projections,
 * and emits change events through the existing SSE infrastructure.
 */
export class LivePollAgent {
  private espnAdapter: ESPNAdapter;
  private standingsAgent: StandingsProjectionAgent;

  constructor(
    private db: Kysely<Database>,
    private logger?: BaseLogger,
  ) {
    this.espnAdapter = new ESPNAdapter(logger);
    this.standingsAgent = new StandingsProjectionAgent(db, logger);
  }

  /**
   * Returns true if current time is within MLB game window.
   * Game hours: roughly 1pm–midnight ET (EDT = UTC-4).
   * In UTC: 17:00–04:00 (next day).
   */
  static isGameWindow(): boolean {
    const h = new Date().getUTCHours();
    return h >= 17 || h < 4;
  }

  /**
   * Refresh standings for a single league from live ESPN data.
   * Returns the number of change events generated.
   */
  async refresh(leagueId: number, today: string): Promise<number> {
    const log = this.logger ?? pino();

    // 1. Load league
    const league = await this.db
      .selectFrom("league")
      .selectAll()
      .where("id", "=", leagueId)
      .executeTakeFirstOrThrow();

    // 2. Fetch ESPN (lightweight — no rosters, no free agents)
    const auth = getEspnAuthFromEnv();
    const espnData = await this.espnAdapter.fetchLeague({
      leagueId: league.espn_league_id,
      season: league.season,
      auth,
      views: ["mSettings", "mTeam", "mStandings"],
    });
    const normalized = this.espnAdapter.normalizeLeagueData(espnData.data);
    const { teams, settings } = normalized;
    const categories = settings.categories;

    if (categories.length === 0) {
      log.warn({ leagueId }, "LivePollAgent: no categories found, skipping");
      return 0;
    }

    // 3. Build ESPN team id → DB team id map
    const dbTeams = await this.db
      .selectFrom("team")
      .select(["id", "espn_team_id"])
      .where("league_id", "=", leagueId)
      .execute();

    if (dbTeams.length === 0) {
      log.warn({ leagueId }, "LivePollAgent: no DB teams found, skipping");
      return 0;
    }

    const espnTeamIdToDbId = new Map(
      dbTeams.map((t) => [t.espn_team_id, t.id]),
    );

    // 4. Get most recent settings_version_id and projections_run_id
    const settingsVersion = await this.db
      .selectFrom("league_settings_version")
      .select("id")
      .where("league_id", "=", leagueId)
      .orderBy("effective_date", "desc")
      .executeTakeFirst();

    const projectionsRun = await this.db
      .selectFrom("projections_run")
      .select("id")
      .where("as_of_date", "<=", today)
      .orderBy("as_of_date", "desc")
      .executeTakeFirst();

    // 5. Ensure today's snapshot row exists (required by StandingsChangeLogAgent)
    if (settingsVersion) {
      await this.db
        .insertInto("league_daily_snapshot")
        .values({
          league_id: leagueId,
          snapshot_date: today,
          created_at: new Date(),
          settings_version_id: settingsVersion.id,
          projections_run_id: projectionsRun?.id ?? null,
          is_degraded: true,
          degraded_reasons: ["live_poll_intraday"],
        })
        .onConflict((oc) =>
          oc.columns(["league_id", "snapshot_date"]).doNothing(),
        )
        .execute();
    }

    // 6. Replace today's category totals, ranks, overall
    await this.db.transaction().execute(async (trx) => {
      // Delete existing
      await trx
        .deleteFrom("team_daily_category_totals")
        .where("league_id", "=", leagueId)
        .where("snapshot_date", "=", today)
        .execute();
      await trx
        .deleteFrom("team_daily_category_ranks")
        .where("league_id", "=", leagueId)
        .where("snapshot_date", "=", today)
        .execute();
      await trx
        .deleteFrom("team_daily_overall")
        .where("league_id", "=", leagueId)
        .where("snapshot_date", "=", today)
        .execute();

      // Insert category totals
      const totalsRows = [];
      for (const team of teams) {
        const dbTeamId = espnTeamIdToDbId.get(team.id);
        if (!dbTeamId) continue;
        for (const cat of categories) {
          totalsRows.push({
            league_id: leagueId,
            snapshot_date: today,
            team_id: dbTeamId,
            category_id: cat.statKey,
            value: team.statValues[cat.espnStatId] ?? 0,
            components_json: null,
          });
        }
      }

      if (totalsRows.length > 0) {
        await trx
          .insertInto("team_daily_category_totals")
          .values(totalsRows)
          .execute();
      }

      // Rank per category and accumulate overall roto points
      const numTeams = teams.length;
      const teamOverallPoints = new Map<number, number>();
      const rankRows = [];

      for (const cat of categories) {
        const teamValues = teams
          .map((t) => ({
            dbTeamId: espnTeamIdToDbId.get(t.id)!,
            value: t.statValues[cat.espnStatId] ?? 0,
          }))
          .filter((tv) => tv.dbTeamId != null);

        teamValues.sort((a, b) =>
          cat.direction === "LOWER_BETTER"
            ? a.value - b.value
            : b.value - a.value,
        );

        const ranked = assignRanksWithTies(
          teamValues.map((tv) => ({ teamId: tv.dbTeamId, value: tv.value })),
          numTeams,
        );

        for (const { teamId, rank, rotoPoints } of ranked) {
          teamOverallPoints.set(
            teamId,
            (teamOverallPoints.get(teamId) ?? 0) + rotoPoints,
          );
          rankRows.push({
            league_id: leagueId,
            snapshot_date: today,
            team_id: teamId,
            category_id: cat.statKey,
            rank,
            roto_points: rotoPoints,
          });
        }
      }

      if (rankRows.length > 0) {
        await trx
          .insertInto("team_daily_category_ranks")
          .values(rankRows)
          .execute();
      }

      // Insert overall
      const overallSorted = Array.from(teamOverallPoints.entries())
        .map(([teamId, value]) => ({ teamId, value }))
        .sort((a, b) => b.value - a.value);

      const overallRanked = assignRanksWithTies(overallSorted, numTeams);

      if (overallRanked.length > 0) {
        await trx
          .insertInto("team_daily_overall")
          .values(
            overallRanked.map(({ teamId, rank }) => ({
              league_id: leagueId,
              snapshot_date: today,
              team_id: teamId,
              overall_rank: rank,
              roto_points: teamOverallPoints.get(teamId) ?? 0,
            })),
          )
          .execute();
      }
    });

    // 7. Recompute projected standings if projections exist
    if (projectionsRun) {
      try {
        await this.standingsAgent.computeProjectedStandings(
          leagueId,
          today,
          projectionsRun.id,
        );
      } catch (err) {
        log.warn(
          { err, leagueId },
          "LivePollAgent: projection recompute failed — non-fatal",
        );
      }
    }

    // 8. Generate change events
    let eventCount = 0;
    try {
      const bus = createNotificationBus(log);
      const changeLogAgent = new StandingsChangeLogAgent(this.db, log, bus);
      eventCount = await changeLogAgent.generateEvents(leagueId, today);
    } catch (err) {
      log.warn(
        { err, leagueId },
        "LivePollAgent: change log generation failed — non-fatal",
      );
    }

    // 9. Emit to SSE clients
    const polledAt = new Date().toISOString();
    if (eventCount > 0) {
      changeLogEmitter.emit(CHANGE_LOG_EVENT, {
        leagueId,
        snapshotDate: today,
        eventCount,
      });
    }
    changeLogEmitter.emit(LIVE_POLL_EVENT, { leagueId, polledAt, eventCount });

    log.info(
      { leagueId, today, eventCount },
      "LivePollAgent: refresh complete",
    );
    return eventCount;
  }
}
```

### Task 3: Create live-poll-worker.ts

New file `packages/backend/src/workers/live-poll-worker.ts`:

```typescript
import { Queue, Worker, type Job } from "bullmq";
import IORedis from "ioredis";
import pino from "pino";
import { getDb } from "../db/connection.js";
import { LivePollAgent } from "../agents/live-poll.js";

const logger = pino({ level: process.env.LOG_LEVEL || "info" });

const connection = new IORedis(
  process.env.REDIS_URL || "redis://localhost:6379",
  { maxRetriesPerRequest: null },
);

export const livePollQueue = new Queue("live-polls", {
  connection: connection as any,
});

interface LivePollJobData {
  leagueId: number;
}

const worker = new Worker<LivePollJobData>(
  "live-polls",
  async (job: Job<LivePollJobData>) => {
    if (process.env.LIVE_POLL_ENABLED !== "true") {
      return { skipped: true, reason: "LIVE_POLL_ENABLED not set" };
    }
    if (!LivePollAgent.isGameWindow()) {
      return { skipped: true, reason: "not in game window" };
    }

    const { leagueId } = job.data;
    const today = new Date().toISOString().split("T")[0];
    const db = getDb();
    const agent = new LivePollAgent(db, logger);

    const eventCount = await agent.refresh(leagueId, today);
    return { leagueId, today, eventCount };
  },
  { connection: connection as any, concurrency: 2 },
);

worker.on("failed", (job, err) => {
  logger.error({ jobId: job?.id, err }, "live-poll job failed");
});

export async function scheduleLivePolls(): Promise<void> {
  if (process.env.LIVE_POLL_ENABLED !== "true") {
    logger.info("LIVE_POLL_ENABLED not set — live poll scheduling skipped");
    return;
  }

  const db = getDb();
  const leagues = await db.selectFrom("league").select("id").execute();

  for (const league of leagues) {
    const jobId = `live-poll-${league.id}`;
    await livePollQueue.add(
      jobId,
      { leagueId: league.id },
      {
        jobId,
        repeat: { pattern: "*/15 * * * *" },
        removeOnComplete: { count: 5 },
        removeOnFail: { count: 10 },
      },
    );
    logger.info({ leagueId: league.id }, "live poll scheduled");
  }
}
```
